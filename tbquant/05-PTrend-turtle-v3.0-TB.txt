    Integer YMD()
    {
        Return Year() * 10000 + Month() * 100 + Day();
    }

    Integer RingCap()
    {
        Return IntPart(Max(60, NPeriod * 3)) + 1;
    }

    // 环尾追加，超出容量从头删除（容量 61，对齐聚宽 attribute_history(61) 窗口）
    Integer PushRing(ArrayRef<Numeric> arr, Numeric v)
    {
        ArrayPushBack(arr, v);
        If (GetArraySize(arr) > RingCap())
        {
            ArrayErase(arr, 0, 1);
        }
        Return 0;
    }

    // 第 i 根日K的真实波幅（i>=1）
    Numeric TRAt(Integer i)
    {
        Numeric t1 = RingDH[i] - RingDL[i];
        Numeric t2 = Abs(RingDH[i] - RingDC[i - 1]);
        Numeric t3 = Abs(RingDL[i] - RingDC[i - 1]);
        Return Max(Max(t1, t2), t3);
    }

    // 复刻聚宽 calc_n：前 NPeriod 个 TR 均值初始化，再按 NSmooth 递推到最新
    Integer RefreshLevels()
    {
        Integer sz = GetArraySize(RingDH);
        If (sz < NPeriod + 1)
        {
            Return 0;
        }
        If (sz < EntryN Or sz < ExitN)
        {
            Return 0;
        }
        Integer i;
        Numeric n = 0;
        For i = 1 To NPeriod
        {
            n = n + TRAt(i);
        }
        n = n / NPeriod;
        For i = NPeriod + 1 To sz - 1
        {
            n = ((NSmooth - 1) * n + TRAt(i)) / NSmooth;
        }
        NVal = n;

        Numeric eh = RingDH[sz - EntryN];
        Numeric el = RingDL[sz - EntryN];
        Numeric xl = RingDL[sz - ExitN];
        Numeric xs = RingDH[sz - ExitN];
        Integer j;
        For j = sz - EntryN + 1 To sz - 1
        {
            eh = Max(eh, RingDH[j]);
            el = Min(el, RingDL[j]);
        }
        For j = sz - ExitN + 1 To sz - 1
        {
            xl = Min(xl, RingDL[j]);
            xs = Max(xs, RingDH[j]);
        }
        EntryLongLv = eh;
        EntryShortLv = el;
        ExitLongLv = xl;
        ExitShortLv = xs;
        Return 0;
    }

    // ---------- 资金分支：回测用 Portfolio_*，实盘/模拟用 A_* ----------

    Bool IsLiveTrade()
    {
        If (StrategyMode() <> Enum_Mode_Strategy_Trade())
        {
            Return False;
        }
        Return A_AccountDataExist(0);
    }

    Numeric CurEquity()
    {
        If (IsLiveTrade())
        {
            Return A_CurrentEquity(0);
        }
        Return Portfolio_CurrentEquity();
    }

    Numeric CurAvailable()
    {
        If (IsLiveTrade())
        {
            Return A_FreeMargin(0);
        }
        Return Portfolio_CurrentCapital();
    }

    // ---------- 手数计算（复刻聚宽 unit_lots + fit_margin + fit_cap）----------

    Integer CalcLots(Numeric px)
    {
        Numeric eq = CurEquity();
        Integer lots = Max(1, IntPart(eq * UnitRisk / (NVal * MultiplierP)));
        Numeric mpl = px * MultiplierP * MarginRateP;
        If (mpl > 0)
        {
            lots = Min(lots, IntPart(CurAvailable() / mpl));
            If (MarginCapPct > 0)
            {
                Numeric room = eq * MarginCapPct - PLots * mpl;
                lots = Min(lots, IntPart(room / mpl));
            }
        }
        If (lots < 0)
        {
            lots = 0;
        }
        Return lots;
    }

    // 开 1 个 unit；台账按预期成交价记账（下单价=触发价，跳空时取开盘价）
    Integer DoOpenUnit(Integer dir, Numeric refPx, String reason)
    {
        Numeric ordPx = refPx;
        If (IsLiveTrade())
        {
            Numeric qp = Q_BidPrice(0);
            If (dir > 0)
            {
                qp = Q_AskPrice(0);
            }
            If (qp > 0)
            {
                ordPx = qp;
            }
        }
        Else
        {
            If (dir > 0 And Open() > refPx)
            {
                ordPx = Open();
            }
            If (dir < 0 And Open() < refPx)
            {
                ordPx = Open();
            }
        }
        Integer lots = CalcLots(Close());
        If (lots < 1)
        {
            // 节流：同一交易日内同一原因只记第一条（3号仓被保证金上限卡住时会逐 bar 重试）
            String skipKey = Text(YMD()) + ":" + reason;
            If (skipKey <> LastSkipKey)
            {
                LastSkipKey = skipKey;
                WLog("OPEN_SKIP 手数不足 reason=" + reason + " dir=" + Text(dir) + " px=" + Text(ordPx, 1));
            }
            Return 0;
        }
        Bool ok = False;
        If (dir > 0)
        {
            ok = Buy(lots, ordPx);
        }
        Else
        {
            ok = SellShort(lots, ordPx);
        }
        If (ok == False)
        {
            WLog("OPEN_FAIL 委托被拒 reason=" + reason + " dir=" + Text(dir) + " lots=" + Text(lots) + " px=" + Text(ordPx, 1));
            Return 0;
        }
        PLots = PLots + lots;
        PCost = PCost + ordPx * lots;
        LastUnitPrice = ordPx;
        UnitCount = UnitCount + 1;
        StopLine = LastUnitPrice - dir * StopDistN * NVal;
        WLog(reason + " OPEN dir=" + Text(dir) + " lots=" + Text(lots) + " px=" + Text(ordPx, 1)
            + " N=" + Text(NVal, 1) + " stop=" + Text(StopLine, 1)
            + " units=" + Text(UnitCount) + " totalLots=" + Text(PLots) + " equity=" + Text(CurEquity(), 0));
        Return 0;
    }

    // 全部平仓（止损/反向突破）；委托失败则保留台账下一 bar 重试
    Integer DoCloseAll(Integer dir, Numeric refPx, String reason)
    {
        If (PLots <= 0)
        {
            Return 0;
        }
        Numeric ordPx = refPx;
        If (IsLiveTrade())
        {
            Numeric qp = Q_AskPrice(0);
            If (dir > 0)
            {
                qp = Q_BidPrice(0);
            }
            If (qp > 0)
            {
                ordPx = qp;
            }
        }
        Else
        {
            If (dir > 0 And Open() < refPx)
            {
                ordPx = Open();
            }
            If (dir < 0 And Open() > refPx)
            {
                ordPx = Open();
            }
        }
        Numeric avg = AvgEntryPrice();
        If (avg <= 0)
        {
            avg = PCost / PLots;
        }
        Integer lots = PLots;
        Bool ok = False;
        If (dir > 0)
        {
            ok = Sell(lots, ordPx);
        }
        Else
        {
            ok = BuyToCover(lots, ordPx);
        }
        If (ok == False)
        {
            WLog("CLOSE_FAIL 委托被拒 reason=" + reason + " dir=" + Text(dir) + " lots=" + Text(lots) + " px=" + Text(ordPx, 1));
            Return 0;
        }
        Numeric pnl = (ordPx - avg) * lots * MultiplierP * dir;
        WLog(reason + " CLOSE dir=" + Text(dir) + " lots=" + Text(lots) + " px=" + Text(ordPx, 1)
            + " avg=" + Text(avg, 1) + " pnl=" + Text(pnl, 0)
            + " units=" + Text(UnitCount) + " equity=" + Text(CurEquity(), 0));
        If (S1Skip > 0 And pnl > 0)
        {
            SkipNextBreakout = True;
            WLog("S1_SKIP 上次突破盈利，下一次同类突破将跳过");
        }
        PLots = 0;
        PCost = 0;
        UnitCount = 0;
        LastUnitPrice = 0;
        StopLine = 0;
        Return 0;
    }

    // ---------- 台账一致性：重启/部分成交后按平台持仓重建 ----------

    Integer RebuildLedger()
    {
        Integer cc = IntPart(Abs(CurrentContracts()));
        PLots = cc;
        UnitCount = Max(1, CurrentEntries());
        Numeric avg = AvgEntryPrice();
        If (avg <= 0)
        {
            avg = Close();
        }
        PCost = avg * cc;
        LastUnitPrice = LastEntryPrice();
        If (LastUnitPrice <= 0)
        {
            LastUnitPrice = avg;
        }
        Integer dir = 1;
        If (MarketPosition() < 0)
        {
            dir = -1;
        }
        StopLine = LastUnitPrice - dir * StopDistN * NVal;
        WLog("STATE_REBUILD cc=" + Text(cc) + " entries=" + Text(UnitCount) + " avg=" + Text(avg, 1)
            + " lastEntry=" + Text(LastUnitPrice, 1) + " stop=" + Text(StopLine, 1));
        Return 0;
    }

    // ---------- 主力移仓跟踪（原生自动换仓后重建基准）----------

    Integer UpdateMainSymbol()
    {
        String ms = MainSymbol();
        If (MainSymInit == False)
        {
            MainSymInit = True;
            LastMainSym = ms;
            Return 0;
        }
        If (ms <> LastMainSym)
        {
            WLog("ROLLOVER " + LastMainSym + " -> " + ms + " mp=" + Text(MarketPosition())
                + " cc=" + Text(CurrentContracts()) + " avg=" + Text(AvgEntryPrice(), 1));
            LastMainSym = ms;
            If (MarketPosition() <> 0)
            {
                RebuildLedger();
            }
        }
        Return 0;
    }

    // ---------- 信号逻辑 ----------

    Bool CanTradeNow()
    {
        If (YMD() < TradeStartDate)
        {
            Return False;
        }
        If (NVal <= 0 Or EntryLongLv <= 0 Or EntryShortLv <= 0 Or ExitLongLv <= 0 Or ExitShortLv <= 0)
        {
            Return False;
        }
        Return True;
    }

    // 入场：突破前 EntryN 日极值（fixed_dir 限定方向；系统1 跳过规则）
    Integer TryEntry()
    {
        Bool allowLong = (FixedDir == 0 Or FixedDir == 1);
        Bool allowShort = (FixedDir == 0 Or FixedDir == -1);
        If (allowLong And High() >= EntryLongLv)
        {
            If (S1Skip > 0 And SkipNextBreakout)
            {
                SkipNextBreakout = False;
                WLog("BREAKOUT_SKIP long level=" + Text(EntryLongLv, 1));
                Return 0;
            }
            DoOpenUnit(1, EntryLongLv, "entry");
            Return 0;
        }
        If (allowShort And Low() <= EntryShortLv)
        {
            If (S1Skip > 0 And SkipNextBreakout)
            {
                SkipNextBreakout = False;
                WLog("BREAKOUT_SKIP short level=" + Text(EntryShortLv, 1));
                Return 0;
            }
            DoOpenUnit(-1, EntryShortLv, "entry");
        }
        Return 0;
    }

    // 每分钟风控与信号：止损 → 反向突破出场 → 加仓 → 入场（每 bar 只执行一个动作）
    Integer TradeLogic()
    {
        If (MarketPosition() == 0)
        {
            If (PLots <> 0 Or UnitCount <> 0)
            {
                WLog("LEDGER_CLEAN 空仓残留台账清理 prevLots=" + Text(PLots));
                PLots = 0;
                PCost = 0;
                UnitCount = 0;
                LastUnitPrice = 0;
                StopLine = 0;
            }
            TryEntry();
            Return 0;
        }
        If (IntPart(Abs(CurrentContracts())) <> PLots)
        {
            RebuildLedger();
        }
        Integer dir = 1;
        If (MarketPosition() < 0)
        {
            dir = -1;
        }
        // 1) 共享止损（2N，含加仓后上移）
        If (StopLine > 0)
        {
            If (dir > 0 And Low() <= StopLine)
            {
                DoCloseAll(dir, StopLine, "stop");
                Return 0;
            }
            If (dir < 0 And High() >= StopLine)
            {
                DoCloseAll(dir, StopLine, "stop");
                Return 0;
            }
        }
        // 2) 反向突破出场
        If (dir > 0 And Low() <= ExitLongLv)
        {
            DoCloseAll(dir, ExitLongLv, "exit_breakout");
            Return 0;
        }
        If (dir < 0 And High() >= ExitShortLv)
        {
            DoCloseAll(dir, ExitShortLv, "exit_breakout");
            Return 0;
        }
        // 3) 加仓：顺向每 AddStepN*N 加 1 unit
        If (UnitCount < MaxUnits)
        {
            If (dir > 0 And High() >= LastUnitPrice + AddStepN * NVal)
            {
                DoOpenUnit(dir, LastUnitPrice + AddStepN * NVal, "add");
                Return 0;
            }
            If (dir < 0 And Low() <= LastUnitPrice - AddStepN * NVal)
            {
                DoOpenUnit(dir, LastUnitPrice - AddStepN * NVal, "add");
                Return 0;
            }
        }
        Return 0;
    }

    OnBar(ArrayRef<Integer> indexs)
    {
        Range[0:DataCount-1]
        {
            BarCnt = BarCnt + 1;

            // ---- 1. 日线聚合：TradingDate 变化 = 上一交易日结束，入环并刷新通道 ----
            TimeStamp td = TradingDate();
            If (TDInit == False)
            {
                TDInit = True;
                CurTD = td;
                TodayH = High();
                TodayL = Low();
                TodayC = Close();
            }
            Else If (td <> CurTD)
            {
                PushRing(RingDH, TodayH);
                PushRing(RingDL, TodayL);
                PushRing(RingDC, TodayC);
                CurTD = td;
                TodayH = High();
                TodayL = Low();
                TodayC = Close();
                RefreshLevels();
            }
            Else
            {
                TodayH = Max(TodayH, High());
                TodayL = Min(TodayL, Low());
                TodayC = Close();
            }

            // ---- 2. 主力移仓检测（原生自动换仓）----
            UpdateMainSymbol();

            // ---- 3. 风控与信号（预热期只建通道，不交易）----
            If (CanTradeNow())
            {
                TradeLogic();
            }

            // ---- 4. 每日定时日志（14:45，同聚宽 signal_time）----
            If (Hour() == LogHour And Minute() == LogMinute And YMD() <> LastLogYmd)
            {
                LastLogYmd = YMD();
                WLog("DAILY " + MainSymbol()
                    + " N=" + Text(NVal, 1) + " EL=" + Text(EntryLongLv, 1) + " XL=" + Text(ExitLongLv, 1)
                    + " ES=" + Text(EntryShortLv, 1) + " XS=" + Text(ExitShortLv, 1)
                    + " units=" + Text(UnitCount) + " lots=" + Text(PLots) + " stop=" + Text(StopLine, 1)
                    + " equity=" + Text(CurEquity(), 0) + " ring=" + Text(GetArraySize(RingDH)));
            }

            // ---- 5. 结束摘要 ----
            If (BarStatus() == 2 And FinalDone == False)
            {
                FinalDone = True;
                WLog("=== 运行结束 BarCnt=" + Text(BarCnt) + " ring=" + Text(GetArraySize(RingDH))
                    + " N=" + Text(NVal, 1) + " mp=" + Text(MarketPosition()) + " cc=" + Text(CurrentContracts())
                    + " lots=" + Text(PLots) + " equity=" + Text(CurEquity(), 0) + " ===");
            }
        }
    }
