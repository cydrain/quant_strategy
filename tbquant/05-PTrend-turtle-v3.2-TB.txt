// v3.2 = v3.1 海龟逻辑（N/四通道/手数/unit/止损/加仓/出场/台账自愈/移仓）原样 + STALL 并行入场。
// 指标与 STALL 口径 = data1 图层周期（默认日线；30m 等日内周期为实验口径，收完即推进，
// 且 Data1[0]/[1] 盘中对齐未复验、与聚宽不可比）。窗口头恒为最新一根已完成 data1 K线，
// 每根新 bar 在 OnBarOpen 刷新一次（实盘形成中 bar 的逐 tick OnBar 不重算）；全局状态随 OnInit 复位。
//
// STALL 并行入场（StallPeriod>0 启用，0=关闭）:
//   事件：某根 data1 bar 创此前 StallPeriod 根新低（多）/新高（空），严格不等；
//   触发：data1 推进后的首根 1m 上，最新完成 bar 距最近事件恰 StallPeriod 根 → 市价开 1 unit
//   （空仓且 FixedDir 允许；与突破并行、先达成先入场；多空同达成多优先）；
//   边沿每根 data1 bar 只评判一次，首次仅建基线；持仓/挂单/预热期达成 = 错过不补。
//   与聚宽差异：聚宽次日 09:01 见新日线，TB 在 data1 推进首根 1m 触发（1d 通常夜盘 21:01），价格/时点不同。
//   窗口上限 2*StallPeriod+5 ≤ WinSize()（默认 61 → 最大 28），超限 OnInit 停用 STALL 并记 ERROR。
//
// 挂载（两图层缺一不可；运行模式「K线收盘模型」）:
//   data0 = 1 分钟：全部信号/风控/下单；data1 = 同合约日线：纯只读指标源。
//   追加：策略单元右键数据源 → 周期 1d → 追加；两图层都「同步为主商品数据范围」。
//   MapToMain=True（默认，模拟/实盘安全口径）：主连图，后复权/映射真实价/自动换仓/委托映射主力；
//   单合约切片回测必须改 False（不复权、不映射，否则委托映射到当前主力）。
//   模拟/实盘校验：data0=data1=MainContSymbol（默认 p9888.DCE，区分大小写与后缀）且 MapToMain=True，
//   否则禁开新仓；OnInit 时 data1 未就绪 → 首个齐备 bar 复核一次（MAPPING_RECHECK）。回测不校验合约。
//   挂单门禁（模拟/实盘）：OnOrder 维护未完成委托集合（Declare/Declared/FillPart/Canceling），
//   非空时拒发任何新委托（开/平/加/损；STALL 边沿遇挂单则放弃本次边沿）；只拦不撤；回测门禁不生效。
//   回调实证（2026-10-19 RateCheckProbe，TB 期货模拟 p2701）：OnOrder 稳定 1→2→5，
//   OnFill 的 side/combOffset/volume/price/fillPrice 有效；fillId 为 String，orderId/fillVolume 待实测。
//   实盘柜台是否回调仍未验证。
//
// 读法红线（MultiSrc 探针实测 43/43 锁定，改代码勿破）:
//   1) N 窗口 = Data1[1..WinSize]，第 WinSize 根仅作前收盘锚点（TR 用 Close[idx+1]）；
//      入场通道 = Data1.High/Low[1..EntryN]；出场通道 = Data1.High/Low[1..ExitN]。
//   2) 信号/风控/下单只读 Data1[1] 及更早。Data1[0] 是禁区（回测=当天完整K=look-ahead，
//      实盘=形成中逐 tick 变）；仅 DAILY 日志 DEBUG 级 d0diag（D1Close0Diag）例外，禁入判断。
//   3) 禁 GetBar（回测返回全 0 Bar）；Data1[1] 恒为最新完成K（日线=昨日；30m=上一根）。
//   4) 信号体只放 Range[0:0] 单块——Range[0:1] 会每图层各跑一遍（早期双图层事故即此，勿复现）。
//   5) Data1 仅出现在 Defs「数据源约定」段访问器内，其余代码一律经访问器；层序变动只改该段一处。
//
// 预热与历史深度【切片5 事故点】：data1 在首个目标信号日前须已有 ≥61 根日线（WinSize=61，≈3 个月），
//   短 1 根都不行——不足时窗口头部越界读数仍是"有效值"，守护不拦截 → N 静默偏高甚至假突破。
//   修复：两图层数据范围同步提前到首个目标信号日前 ≥61 个交易日再跑。
//   读取确为无效值/0 时守护才生效（ERROR、拒下单、每根新 bar 重试；日志按交易日节流）。
//

Defs
    // ---------- 基础辅助 ----------

    // 日志目录规范化：非空且末位不是 / 时自动补 /（参数漏写结尾斜杠不再拼坏文件名）
    String EffectiveLogDir() {
        If (LogDir == "") {
            Return "";
        }
        If (Right(LogDir, 1) == "/") {
            Return LogDir;
        }
        Return LogDir + "/";
    }

    // 回测模式 = 既非交易也非图表（简单回测实测 StrategyMode=4；图表模式=0 另算）。
    Bool IsBacktestMode() {
        Return StrategyMode() <> Enum_Mode_Strategy_Trade() And StrategyMode() <> Enum_Mode_Strategy_Chart();
    }

    // 日志路径：有效日志目录（可空=平台运行目录）+ 公式名 + data0 合约 + [非交易 _BT] + 平台当前日期
    // （实盘按物理日分文件；非交易模式：简单回测/图表历史计算/图表回测，统一写 _BT 文件与模拟隔离；
    //   平台不区分“图表回测”和“模拟前历史重算”，两者都写、都不删；P/RU 多实例同目录也不混）
    String LogPath() {
        String modeTag = "";
        If (StrategyMode() <> Enum_Mode_Strategy_Trade()) {
            modeTag = "_BT";
        }
        Return EffectiveLogDir() + FormulaName() + "_" + Symbol + modeTag + "_日志_" + Text(CurrentDate());
    }

    // 时间戳：WLog*（bar 上下文）用 bar 日期时间；OnInit 阶段无 bar 时间 → 只能用 WLogInit*，取平台当前时间
    String BarStamp() {
        Return Text(Date()) + " " + Text(Time());
    }

    String InitStamp() {
        Return DateTimeToString(CurrentDate());
    }

    // 日志核心：Param LogLevel 阈值过滤（消息级别 ≤ LogLevel 才写出），行首标记=级别：
    //   !!! [Error] !!! 策略无法按预期工作（数据缺失/委托被拒） · [Warn] 异常已自愈/机会丢失/换月
    //   [Info] 正常心跳与成交 · [Debug] 诊断细节
    //   （.tbf 为 TB 加密日志文件（非纯文本），无颜色通道，ERROR 以 !!! [Error] !!! 醒目化替代红色）
    // 双写：文件 +（PrintToUI=True 时）K 线图控制台，同一阈值。
    // 模式：非交易（简单回测/图表历史计算/图表回测）→*_BT_日志_*；模拟/实盘→*_日志_*；所有模式都写。
    Integer WLogCore(Integer need, String tag, String ts, String s) {
        If (LogLevel >= need) {
            String msg = ts + " | " + tag + " " + s;
            FileAppend(LogPath(), msg, False);
            If (PrintToUI) {
                Print(msg);
            }
        }
        Return 0;
    }

    Integer WLogE(String s) {
        Return WLogCore(0, "!!! [Error] !!!", BarStamp(), s);
    }

    Integer WLogW(String s) {
        Return WLogCore(1, "[Warn]", BarStamp(), s);
    }

    Integer WLog(String s) {
        Return WLogCore(2, "[Info]", BarStamp(), s);
    }

    Integer WLogD(String s) {
        Return WLogCore(3, "[Debug]", BarStamp(), s);
    }

    Integer WLogInitE(String s) {
        Return WLogCore(0, "!!! [Error] !!!", InitStamp(), s);
    }

    Integer WLogInit(String s) {
        Return WLogCore(2, "[Info]", InitStamp(), s);
    }

    // 回报事件日志：事件在 bar 上下文外驱动，时间戳取平台时钟（Date()/Time() 可能为旧值）
    Integer WLogEvt(String s) {
        Return WLogCore(2, "[Event]", DateTimeToString(SystemDateTime()), s);
    }

    Integer YMD() {
        Return Year() * 10000 + Month() * 100 + Day();
    }

    // 数值有效性：InvalidNumeric（≈1.79e308）或 <=0 均为无效。
    // 成对合取，不依赖 And 短路；用于 OHLC、N、委托价等逐值校验。
    Bool NumValid(Numeric value) {
        Return (value != InvalidNumeric And value > 0);
    }

    // 刷新失败日志（按交易日节流）：失败态每根新 bar 都会重试——每天只报首条防刷屏
    Integer WLogRefreshErr(String s) {
        If (YMD() <> LastRefreshErrYmd) {
            LastRefreshErrYmd = YMD();
            WLogE(s);
        }
        Return 0;
    }

    // 日志节流状态复位（OnInit 调用）：0 与当年任何日期不等 → 每条日志首条必记
    Integer ResetLogThrottle() {
        LastLogYmd = 0;
        LastSkipYmd = 0;
        LastRefreshErrYmd = 0;
        LastGateYmd = 0;
        LastBadPxYmd = 0;
        LastAcctWarnYmd = 0;
        Return 0;
    }

    // 窗口根数（对齐聚宽 attribute_history(need+1) 与 v3.0 环容量）；STALL 计数窗口 2*StallPeriod+5 必须 ≤ 本值
    Integer WinSize() {
        Return IntPart(Max(60, NPeriod * 3)) + 1;
    }

    // ---------- 参数校验（OnInit 调用一次）----------
    // 核心参数非法 → 只禁开新仓/加仓，止损与平仓路径不受影响。
    // 计算口径参数（NPeriod/NSmooth/EntryN/ExitN/窗口）另汇总为 CalcOk：
    // 非法时 RefreshNChannels 直接保持五值 0，避免带病计算。

    // 正常返回，设置仍生效）；SlipTicks 仍要求 >=0。
    Integer ParamErr(String name, String value, String rule) {
        WLogInitE("参数非法: " + name + "=" + value + "，要求 " + rule);
        Return 0;
    }

    Bool ValidateParams() {
        Bool ok = True;
        If (FixedDir <> -1 And FixedDir <> 0 And FixedDir <> 1) {
            ParamErr("FixedDir", Text(FixedDir), "-1/0/1");
            ok = False;
        }
        If (NPeriod < 1) {
            ParamErr("NPeriod", Text(NPeriod), ">=1");
            ok = False;
        }
        If (NSmooth < 1) {
            ParamErr("NSmooth", Text(NSmooth), ">=1");
            ok = False;
        }
        If (EntryN < 1) {
            ParamErr("EntryN", Text(EntryN), ">=1");
            ok = False;
        }
        If (ExitN < 1) {
            ParamErr("ExitN", Text(ExitN), ">=1");
            ok = False;
        }
        If (WinSize() < EntryN Or WinSize() < ExitN) {
            ParamErr("WinSize", Text(WinSize()), ">=EntryN(" + Text(EntryN) + ") 且 >=ExitN(" + Text(ExitN) + ")");
            ok = False;
        }
        If (UnitRisk <= 0) {
            ParamErr("UnitRisk", Text(UnitRisk, 4), ">0");
            ok = False;
        }
        If (MarginCapPct < 0) {
            ParamErr("MarginCapPct", Text(MarginCapPct, 4), ">=0（0=不限）");
            ok = False;
        }
        If (MaxUnits < 1) {
            ParamErr("MaxUnits", Text(MaxUnits), ">=1");
            ok = False;
        }
        If (AddStepN <= 0) {
            ParamErr("AddStepN", Text(AddStepN, 2), ">0");
            ok = False;
        }
        If (StopDistN <= 0) {
            ParamErr("StopDistN", Text(StopDistN, 2), ">0");
            ok = False;
        }
        If (InitCapital <= 0) {
            ParamErr("InitCapital", Text(InitCapital, 0), ">0");
            ok = False;
        }
        If (MultiplierP <= 0) {
            ParamErr("MultiplierP", Text(MultiplierP, 2), ">0");
            ok = False;
        }
        If (MarginRateP <= 0) {
            ParamErr("MarginRateP", Text(MarginRateP, 4), ">0");
            ok = False;
        }
        If (CommissionRateP < 0) {
            ParamErr("CommissionRateP", Text(CommissionRateP, 8), ">=0");
            ok = False;
        }
        If (SlipTicks < 0) {
            ParamErr("SlipTicks", Text(SlipTicks, 2), ">=0");
            ok = False;
        }
        If (TradeStartDate < 19000101 Or TradeStartDate > 21001231) {
            ParamErr("TradeStartDate", Text(TradeStartDate), "yyyyMMdd（19000101..21001231）");
            ok = False;
        }
        CalcOk = (NPeriod >= 1 And NSmooth >= 1 And EntryN >= 1 And ExitN >= 1
            And WinSize() >= EntryN And WinSize() >= ExitN);
        Return ok;
    }

        Return Data1.Close[0];
    }

    // 日线第 idx 根的真实波幅（idx>=1；用 idx 与 idx+1 两根，第 idx+1 根作前收盘锚点）
    // 写法=探针验证版直读的访问器封装（语义等价）；禁 GetBar。
    Numeric D1TR(Integer idx) {
        Numeric t1 = D1High(idx) - D1Low(idx);
        Numeric t2 = Abs(D1High(idx) - D1Close(idx + 1));
        Numeric t3 = Abs(D1Low(idx) - D1Close(idx + 1));
        Return Max(Max(t1, t2), t3);
    }

    // N：经典海龟口径（与聚宽 calc_n 同式）：前 NPeriod 个 TR 均值初始化，
    // 再按 N=((NSmooth-1)*N+TR)/NSmooth 递推到最新。
    // 窗口 Data1[1..win]（base=1）；第 win 根仅作前收盘锚点。
    // 与 v3.0 环口径探针实测逐值全等（M3 43/43 MATCH 61@base1）。调用前须通过 RefreshNChannels 守卫。
    Numeric CalcD1N(Integer win) {
        Integer t;
        Numeric n = 0;
        For t = 1 To NPeriod {
            n = n + D1TR(win - t);
        }
        n = n / NPeriod;
        For t = NPeriod + 1 To win - 1 {
            n = ((NSmooth - 1) * n + D1TR(win - t)) / NSmooth;
        }
        // NaN 守卫：NPeriod/NSmooth ≤0 等参数滥用产生除零 → 无效值/无穷；
        // 无效值会使 CanTradeNow 的 NVal<=0 比较恒 False 而漏过门禁 → 此处归一为 0
        //（RefreshNChannels 见 0 记 ERROR 并保持五值全 0，拒绝交易、每根新 bar 自动重试；错误日志按交易日节流）
        If (NumValid(n)) {
            Return n;
        }
        Return 0;
    }

    // 四通道（唐奇安）：Data1[1]=最近一根完整日K，其后按各自周期数取极值。
    // 写入 EntryLongLv/EntryShortLv/ExitLongLv/ExitShortLv；调用前须通过 RefreshNChannels 守卫。
    Integer CalcChannels() {
        Numeric eh = D1High(1);
        Numeric el = D1Low(1);
        Numeric xl = D1Low(1);
        Numeric xs = D1High(1);
        Integer j;
        For j = 2 To EntryN {
            eh = Max(eh, D1High(j));
            el = Min(el, D1Low(j));
        }
        For j = 2 To ExitN {
            xl = Min(xl, D1Low(j));
            xs = Max(xs, D1High(j));
        }
        EntryLongLv = eh;
        EntryShortLv = el;
        ExitLongLv = xl;
        ExitShortLv = xs;
        Return 0;
    }

    // N 与四通道复位为无效值（0）：刷新「先清后算」与 OnInit 复位共用
    Integer ResetChannels() {
        NVal = 0;
        EntryLongLv = 0;
        EntryShortLv = 0;
        ExitLongLv = 0;
        ExitShortLv = 0;
        Return 0;
    }

    // N 与四通道的刷新入口（编排器：零化 → 守卫 → 计算，数学都在被调函数里）。
    // 概念映射：NVal ← CalcD1N(win)（经典海龟 N：前 NPeriod 个 TR 均值种子 + Wilder 递推）；
    //           四通道 ← CalcChannels()（唐奇安 EntryN 入场 / ExitN 出场极值）。
    // 调用时机：OnBarOpen（每根新 bar 生成时一次；正常历史回放与实盘均驱动）——实盘 OnBar
    //           会随形成中 bar 逐 tick 重复，故刷新不放 OnBar；TryStallEntry 消费新 data1 边沿时
    //           补刷一次（防 OnBarOpen 仍读到旧 Data1[1]）。值随 data1 窗口头推进更新：
    //           Data1[1]=最新一根已完成 data1 K线，日线仅换日推进（探针 M3 实测 43/43 对齐
    //           v3.0 环）——同一根 K 期间与旧版「换日一刷」取值一致；30 分钟等日内周期每根收完即推进。
    // 数据窗口：data1 图层经 D1* 访问器按 base=1 取 win=WinSize() 根，第 win 根仅作前收盘锚点。
    // 守卫顺序：① CalcOk（OnInit 参数校验：NPeriod/NSmooth/EntryN/ExitN/窗口）
    //           ② DataCount≥2（有 data1 图层）③ data1 与 data0 同合约（错位拒交易，
    //           防拿错合约的 N/通道下单）④ D1Close(1)/(win) 有效且 >0（历史深度）。
    // 失败语义：五值先全部置 0；CalcOk=False → 静默保持 0（OnInit 已记 ERROR）；
    //           其余守卫不过 → 保持 0 + ERROR（按日节流）→ CanTradeNow 拒绝，下根新 bar 自动重试。
    // 后置保证：成功返回时 NVal 与四通道均 >0；CalcD1N/CalcChannels 只允许本函数在守卫通过后调用。
    Integer RefreshNChannels() {
        Integer win = WinSize();
        ResetChannels();
        // 计算口径参数非法（OnInit 已记 ERROR）：五值保持 0，避免带病计算
        If (CalcOk == False) {
            Return 0;
        }
        If (DataCount() < 2) {
            WLogRefreshErr("N/四通道刷新失败 NO Data1 source, DataCount=" + Text(DataCount()) + "（需追加同合约 data1 图层）");
            Return 0;
        }
        If (D1Sym() <> Symbol) {
            WLogRefreshErr("N/四通道刷新失败 data1.Symbol≠data0.Symbol, data1=" + D1Sym() + " data0=" + Symbol + "（图层顺序/合约不符，检查图表数据源）");
            Return 0;
        }
        // 历史深度守卫：根数不足时序列读取返回无效值（官方：数据不足→无效值）。
        // NumValid 同时排除 InvalidNumeric 与 <=0，不依赖 And 短路。
        Bool valid = NumValid(D1Close(1)) And NumValid(D1Close(win));
        If (valid == False) {
            WLogRefreshErr("N/四通道刷新失败 D1 history too short, win=" + Text(win)
                + " Close[1]=" + IIFString(D1Close(1) != InvalidNumeric, "ok", "invalid")
                + " Close[win]=" + IIFString(D1Close(win) != InvalidNumeric, "ok", "invalid")
                + "（data1 历史不足或缺失）");
            Return 0;
        }
        // 前置守卫已通过 → 窗口内各 bar 有效，交计算函数
        NVal = CalcD1N(win);
        If (NVal <= 0) {   // CalcD1N 已做 NaN 守卫（参数滥用归 0）；此处保持「失败=全 0 + ERROR」语义，CalcChannels 不执行
            WLogRefreshErr("N/四通道刷新失败 CalcD1N 无效, NVal=" + Text(NVal, 2) + "（检查 NPeriod/NSmooth 参数）");
            Return 0;
        }
        CalcChannels();
        Return 0;
    }

    // ---------- 资金分支：回测用 Portfolio_*，实盘/模拟用 A_* ----------

    // 策略交易模式（模拟/实盘图表）：不等同于账户已就绪，账户数据可能晚于 OnInit 才就绪。
    Bool IsStrategyTradeMode() {
        Return StrategyMode() == Enum_Mode_Strategy_Trade();
    }

    // 实盘/模拟判定（全部资金读取的岔路口）：策略交易模式 + 0 号账户数据已就绪。
    // 第二条件防"模式对了但账户未连线/未就绪"；回测下 A_AccountDataExist=False（探针 R6 实测）。
    Bool IsLiveTrade() {
        If (IsStrategyTradeMode() == False) {
            Return False;
        }
        Return A_AccountDataExist(0);
    }

    // 当前阶段是否允许发单（True=可交易）。只管「阶段」；账户/日期/通道等业务条件由 CanTradeNow 判断。
    // 简单回测/参数优化(mode≠0,1)=全部 bar 可交易；模拟/实盘(mode=1)=仅实时 bar；
    // 图表模式(mode=0)=自动交易关（图表回测）可交易；自动交易开（图表交易）一律不发单。
    Bool IsTradablePhase() {
        Bool inBacktest = IsBacktestMode();      // 简单回测/参数优化：mode≠Trade 且 ≠Chart
        Bool inTrade = IsStrategyTradeMode();    // 模拟/实盘策略交易：mode=Trade
        Bool isRealtimeBar = (QuoteStatus() == Enum_QuoteStatus_RealTime);

        // 1) 简单回测/参数优化（mode≠0,1）：每根 bar 都参与模拟成交
        If (inBacktest) {
            Return True;
        }
        // 2) 模拟/实盘（mode=1）：只允许当前实时 bar
        If (inTrade) {
            Return isRealtimeBar;
        }
        // 3) 图表模式（mode=0）：自动交易关=图表回测→可交易；自动交易开=图表交易→一律不发单
        Return IsTradeEnabled() == False;
    }

    // 总权益（含浮盈）：实盘→A_CurrentEquity(0)；回测→Portfolio_CurrentEquity()。供权益日志。
    Numeric CurEquity() {
        If (IsLiveTrade()) {
            Return A_CurrentEquity(0);
        }
        Return Portfolio_CurrentEquity();
    }

    // 风险资金：Min(InitCapital, 当前权益)，回测/实盘同构——盈利超过参数不放大手数，
    // 亏损时按权益缩小。供 CalcLots 风险段与占用上限。
    Numeric RiskCapital() {
        If (IsLiveTrade()) {
            Return Min(InitCapital, A_CurrentEquity(0));
        }
        Return Min(InitCapital, Portfolio_CurrentEquity());
    }

    // 可用资金：实盘→A_FreeMargin(0)（空闲保证金）；回测→Portfolio_CurrentCapital()。
    // 供 CalcLots 资金上限。
    Numeric CurAvailable() {
        If (IsLiveTrade()) {
            Return A_FreeMargin(0);
        }
        Return Portfolio_CurrentCapital();
    }

    // ---------- 手数计算（复刻聚宽 unit_lots + fit_margin + fit_cap）----------

    // 开仓手数计算：三段连续削峰（风险定量 → 资金 → 占用上限），逐段只能更小。
    // 风险基数 = Min(InitCapital, 当前权益)；② 用实际可用资金。
    // 唯一调用点 DoOpenUnitCore（传当前价 Close()）；price 仅用于估算每手保证金。
    Integer CalcLots(Numeric price) {
        Numeric eq = RiskCapital();
        // ① 经典海龟 unit：1N 波动 ≈ 风险资金 × UnitRisk（1%），向下取整、最低 1 手
        Integer lots = Max(1, IntPart(eq * UnitRisk / (NVal * MultiplierP)));
        Numeric marginPerLot = price * MultiplierP * MarginRateP;   // 每手保证金
        If (marginPerLot > 0) {   // ≤0 时跳过②③（防除零；聚宽同款守卫）
            // ② 物理约束：可用资金 ÷ 每手保证金（聚宽 fit_lots_to_margin）
            lots = Min(lots, IntPart(CurAvailable() / marginPerLot));
            If (MarginCapPct > 0) {   // =0 表示不限
                // ③ 占用上限：剩余额度 = 风险资金 × MarginCapPct − 已持仓占用（聚宽 fit_lots_to_cap）
                Numeric room = eq * MarginCapPct - PLots * marginPerLot;
                lots = Min(lots, IntPart(room / marginPerLot));
            }
        }
        Return IIF(lots < 0, 0, lots);   // ②③ 可削到负 → 归 0，语义 = 开不出
    }

    // ---------- 订单封装 ----------

    // 委托价解析：理论触发价 refPrice → 实际可成交价。
    // orderDir = 委托方向（+1 买 / -1 卖；平仓调用方传入持仓方向取反）。
    // 实盘：买吃卖一、卖吃买一（跨价差保成交）；盘口无效(≤0，如未连线/休市)退回 refPrice。
    // 回测：开盘跳空越过触发价时按开盘价成交（对委托方向更不利的价），否则按 refPrice。
    Numeric CalcOrderPrice(Integer orderDir, Numeric refPrice) {
        If (IsLiveTrade()) {
            Numeric qp = Q_BidPrice(0);   // 卖方向默认取买一
            If (orderDir > 0) {
                qp = Q_AskPrice(0);   // 买方向改取卖一（对手价，跨价差保成交）
            }
            If (NumValid(qp)) {   // 盘口无效时 Q_* 返回 InvalidNumeric（≈1.79e308）或 0
                Return qp;
            }
            Return refPrice;   // 盘口无效（未连线/休市/历史回放）→ 理论价兜底
        }
        If (orderDir > 0 And Open() > refPrice) {   // 高开越过触发价：买单按开盘价（对买方更贵）
            Return Open();
        }
        If (orderDir < 0 And Open() < refPrice) {   // 低开越过触发价：卖单按开盘价（对卖方更低）
            Return Open();
        }
        Return refPrice;   // 无跳空：按触发价
    }

    // ---------- 挂单门禁（模拟/实盘）----------
    // OnOrder 是唯一能拿到未完成委托的回报入口（A_SendOrderEx 可直取 orderId，本策略用
    // Buy/SellShort 只能靠事件）。活跃状态入集合，其余状态（含未知）一律视为终态移除，
    // 避免状态枚举扩展时出现永不解除的死锁。回测不触发事件 → 集合恒空。
    Bool OrderIsPending(Integer st) {
        Return (st == Enum_Declare Or st == Enum_Declared Or st == Enum_FillPart Or st == Enum_Canceling);
    }

    Integer PendingIndex(Integer oid) {
        Integer n = GetArraySize(PendingIds);
        Integer i;
        For i = 0 To n - 1 {
            If (IntPart(PendingIds[i]) == oid) {
                Return i;
            }
        }
        Return -1;
    }

    Integer PendingAdd(Integer oid) {
        If (PendingIndex(oid) < 0) {
            ArrayPushBack(PendingIds, oid);
        }
        Return 0;
    }

    Integer PendingRemove(Integer oid) {
        Integer idx = PendingIndex(oid);
        If (idx >= 0) {
            ArrayErase(PendingIds, idx, 1);
        }
        Return 0;
    }

    Integer PendingClear() {
        Integer n = GetArraySize(PendingIds);
        Integer i;
        For i = 1 To n {
            ArrayErase(PendingIds, 0, 1);
        }
        Return 0;
    }

    String PendingStr() {
        String s = "";
        Integer n = GetArraySize(PendingIds);
        Integer i;
        For i = 0 To n - 1 {
            If (s <> "") {
                s = s + ",";
            }
            s = s + Text(IntPart(PendingIds[i]));
        }
        Return s;
    }

    // 发送委托：开仓/平仓共用入口；失败记 FAIL 日志并返回 False（调用方放弃本次动作）。
    // orderDir = 委托方向（+1 买 / -1 卖）；isClose=True 平仓（卖平/买平），False 开仓（买开/卖开）。
    // 日志中 dir= 恒为委托方向（平仓被拒时与持仓方向符号相反，便于对照委托本身）。
    // 注意：分发必须用 If 块——IIF 两侧都会求值，会把两种指令都发出去。
    Bool DoSendOrder(Integer orderDir, Integer lots, Numeric price, Bool isClose, String reason) {
        // 无效委托价兜底：NumValid 不通过（InvalidNumeric≈1.79e308 或 <=0）一律不发单——
        // 防历史回放/盘口未就绪时把无效价送进交易层（TBCore: trade Error Args）
        If (NumValid(price) == False) {
            If (LastBadPxYmd != YMD()) {
                LastBadPxYmd = YMD();
                WLogW("ORDER_SKIP 无效委托价 reason=" + reason
                    + " dir=" + Text(orderDir) + " lots=" + Text(lots) + " price=" + Text(price, 1));
            }
            Return False;
        }
        // 挂单门禁：存在未完成委托 → 一律不发新单（开/平/加/损同规则；回测集合恒空）
        If (GetArraySize(PendingIds) > 0) {
            If (LastGateYmd <> YMD()) {
                LastGateYmd = YMD();
                WLogW("ORDER_GATE 未完成委托=" + Text(GetArraySize(PendingIds))
                    + " ids=" + PendingStr() + "，禁发新单 reason=" + reason
                    + " dir=" + Text(orderDir) + " lots=" + Text(lots)
                    + " close=" + IIFString(isClose, "True", "False"));
            }
            Return False;
        }
        Bool ok = False;
        If (isClose) {
            If (orderDir > 0) {
                ok = BuyToCover(lots, price);
            }
            Else {
                ok = Sell(lots, price);
            }
        }
        Else {
            If (orderDir > 0) {
                ok = Buy(lots, price);
            }
            Else {
                ok = SellShort(lots, price);
            }
        }
        If (ok == False) {
            WLogE(IIFString(isClose, "CLOSE_FAIL", "OPEN_FAIL") + " 委托被拒 reason=" + reason
                + " dir=" + Text(orderDir) + " lots=" + Text(lots) + " price=" + Text(price, 1));
        }
        Return ok;
    }

    // 开 1 个 unit 的公共实现：价格已解析；台账按预期成交价记账
    // 开仓/加仓唯一出口：ParamsOk=False（OnInit 参数校验失败）在此兜底拒绝
    Integer DoOpenUnitCore(Integer dir, Numeric orderPrice, String reason) {
        If (ParamsOk == False) {
            Return 0;
        }
        Integer lots = CalcLots(Close());
        If (lots < 1) {
            // 节流：手数不足当日不会改善（逐 bar 重试均被拒），每天只记第一条防刷屏
            If (LastSkipYmd <> YMD()) {
                LastSkipYmd = YMD();
                WLogW("OPEN_SKIP 手数不足 reason=" + reason + " dir=" + Text(dir) + " price=" + Text(orderPrice, 1));
            }
            Return 0;
        }
        If (DoSendOrder(dir, lots, orderPrice, False, reason) == False) {
            Return 0;
        }
        PLots = PLots + lots;
        PCost = PCost + orderPrice * lots;
        LastUnitPrice = orderPrice;
        UnitCount = UnitCount + 1;
        StopLine = LastUnitPrice - dir * StopDistN * NVal;
        WLog(reason + " OPEN dir=" + Text(dir) + " lots=" + Text(lots) + " price=" + Text(orderPrice, 1)
            + " N=" + Text(NVal, 1) + " stop=" + Text(StopLine, 1)
            + " units=" + Text(UnitCount) + " totalLots=" + Text(PLots) + " equity=" + Text(CurEquity(), 0));
        Return 0;
    }

    // 触发价入场（突破/加仓）：理论触发价 refPrice → CalcOrderPrice 解析实际可成交价
    Integer DoOpenUnit(Integer dir, Numeric refPrice, String reason) {
        Return DoOpenUnitCore(dir, CalcOrderPrice(dir, refPrice), reason);
    }

    // 市价入场（STALL 边沿）：回测=当前 1m 收盘价；实盘=盘口对手价（无效回退 Close）
    Integer DoOpenUnitMarket(Integer dir, String reason) {
        Numeric orderPrice = Close();
        If (IsLiveTrade()) {
            orderPrice = CalcOrderPrice(dir, Close());
        }
        Return DoOpenUnitCore(dir, orderPrice, reason);
    }

    // 台账清零：OnInit 复位 / 平仓成功后 / 空仓残留清理共用
    Integer ResetLedger() {
        PLots = 0;
        PCost = 0;
        UnitCount = 0;
        LastUnitPrice = 0;
        StopLine = 0;
        Return 0;
    }

    // 全部平仓（止损/反向突破）；委托失败则保留台账下一 bar 重试
    Integer DoCloseAll(Integer dir, Numeric refPrice, String reason) {
        If (PLots <= 0) {
            Return 0;
        }
        // 委托方向 = 持仓方向取反（多平=卖、空平=买）
        Numeric orderPrice = CalcOrderPrice(dir * -1, refPrice);
        Numeric avg = AvgEntryPrice();
        If (avg <= 0) {
            avg = PCost / PLots;
        }
        Integer lots = PLots;
        If (DoSendOrder(dir * -1, lots, orderPrice, True, reason) == False) {
            Return 0;
        }
        Numeric pnl = (orderPrice - avg) * lots * MultiplierP * dir;
        WLog(reason + " CLOSE dir=" + Text(dir) + " lots=" + Text(lots) + " price=" + Text(orderPrice, 1)
            + " avg=" + Text(avg, 1) + " pnl=" + Text(pnl, 0)
            + " units=" + Text(UnitCount) + " equity=" + Text(CurEquity(), 0));
        If (S1Skip > 0 And pnl > 0) {
            SkipNextBreakout = True;
            WLog("S1_SKIP 上次突破盈利，下一次同类突破将跳过");
        }
        ResetLedger();
        Return 0;
    }

    // ---------- 台账一致性：重启/部分成交后按平台持仓重建 ----------

    // 台账重建：唯一「以平台为准」的写入口——乐观台账（下单成功即记账）与平台失同步时整体覆盖。
    // 触发场景：① ManagePosition 每 bar 数量对账失配（重启全局归零 / 部分成交 / 手工干预等）；
    //          ② 换月——UpdateMainSymbol 检测到主力变化且持仓≠0（基准换到新合约）。
    // 数据映射：PLots ← CurrentContracts；
    //          UnitCount ← CurrentEntries（下限 1）；
    //          PCost ← AvgEntryPrice×cc（≤0 退回收盘价）；
    //          LastUnitPrice ← LastEntryPrice（≤0 退回均价）；
    //          StopLine ← LastUnitPrice − dir×StopDistN×NVal（按海龟规则再推导）。
    // 局限：StopLine 用的是重建时的 NVal，非同 unit 入场时值——ATR 已变则偏差 ≈ 2×ΔN；
    //       仅数量失配触发——发单价 vs 成交价的纯价格偏差不在此清除（留待下次数量失配顺带重建）。
    // 后置保证：返回后 PLots 与平台手数一致（守卫收敛，不会连续重建）；调用点均在意向判断
    //           （止损/加仓/入场）之前，同 bar 风控即使用修复后的台账。
    Integer RebuildLedger() {
        // 平台真值：当前总手数（Abs 统一多空符号、IntPart 取整）
        Integer cc = IntPart(Abs(CurrentContracts()));
        PLots = cc;
        // 开仓次数 = unit 数（每次加仓是独立 entry）；Max 兜底：有持仓至少 1
        UnitCount = Max(1, CurrentEntries());
        // 平台成交加权均价；未就绪/异常（≤0）退回收盘价，避免 PCost 归零
        Numeric avg = AvgEntryPrice();
        If (avg <= 0) {
            avg = Close();
        }
        PCost = avg * cc;
        // 平台末次开仓成交价（加仓步进与共享止损的基准）；异常（≤0）退回均价
        LastUnitPrice = LastEntryPrice();
        If (LastUnitPrice <= 0) {
            LastUnitPrice = avg;
        }
        // 共享止损再推导：末 unit ± StopDistN×N（随持仓方向；用重建时的 N，见头注局限）
        Integer dir = 1;
        If (MarketPosition() < 0) {
            dir = -1;
        }
        StopLine = LastUnitPrice - dir * StopDistN * NVal;
        // WARNING 留痕：contracts/entries/avg/lastEntry/stop 全量落日志，便于事后审计
        WLogW("STATE_REBUILD contracts=" + Text(cc) + " entries=" + Text(UnitCount) + " avg=" + Text(avg, 1)
            + " lastEntry=" + Text(LastUnitPrice, 1) + " stop=" + Text(StopLine, 1));
        Return 0;
    }

    // ---------- 主力移仓跟踪（原生自动换仓后重建基准）----------

    // 主力换月检测：比较 MainSymbol() 字符串，变化=换月——留 ROLLOVER 日志，持仓≠0 时重建台账
    // 基准（TB 原生自动换仓把平台均价换到新合约价格口径，策略台账需跟随重建）。
    // 调用时机：OnBar 每 bar 无条件（不放 ManagePosition 内——空仓时换月日志也要记；也不放 OnBarOpen——
    //           重建须紧贴交易判断：bar 中途换月时同一 OnBar 内先重建、后风控，旧口径 StopLine 不漏判）。
    // 盲区：停机期间发生换月时，重启后 MainSymInit 走静默分支不报 ROLLOVER——一致性由
    //       RebuildLedger 的数量守卫兜底，仅缺日志条（增强备选：按换月日历时间兜底检测）。
    // 后置保证：返回后 LastMainSym == 当前 MainSymbol()，同一换月不会重复触发。
    Integer UpdateMainSymbol() {
        // 平台当前主力合约代码（主连/映射模式下随换月切换）
        String ms = MainSymbol();
        // 初始化分支：首个 bar 只记基线——重启后 LastMainSym 可能残留上一轮值，
        // MainSymInit=False 跳过比较，防启动瞬间误报
        If (MainSymInit == False) {
            MainSymInit = True;
            LastMainSym = ms;
            Return 0;
        }
        // 字符串变化 = 发生换月（自动换仓已在平台侧完成）
        If (ms <> LastMainSym) {
            // WARNING 留痕：旧→新 + 换仓后的平台状态（pos=-1/0/1，contracts=平台手数）
            WLogW("ROLLOVER " + LastMainSym + " -> " + ms + " pos=" + Text(MarketPosition())
                + " contracts=" + Text(CurrentContracts()) + " avg=" + Text(AvgEntryPrice(), 1));
            LastMainSym = ms;
            // 有持仓：台账基准整体换成新合约口径（均价/末次入场价/止损线）
            If (MarketPosition() <> 0) {
                RebuildLedger();
            }
        }
        Return 0;
    }

    // ---------- 信号逻辑 ----------

    // OnInit 时 data1 可能尚未就绪 → MappingOk 误锁。首个 data1 齐备的 bar 复核一次：
    // 数据层就绪后仍不符（合约错位/非主连/MapToMain=False）则保持锁定并记 ERROR（静态配置，不会运行中变化）。
    Integer RecheckMapping() {
        If (MappingRecheck == False) {
            Return 0;
        }
        If (StrategyMode() <> Enum_Mode_Strategy_Trade()) {
            MappingRecheck = False;
            Return 0;
        }
        If (DataCount() < 2) {
            Return 0;   // data1 尚未挂载/就绪，下一 bar 再试
        }
        MappingRecheck = False;
        Bool isMainCont = (Symbol == MainContSymbol);
        Bool layerOk = (D1Sym() == Symbol);
        If (isMainCont And layerOk And MapToMain) {
            MappingOk = True;
            WLog("MAPPING_RECHECK ok=True data0=" + Symbol + " data1=" + D1Sym()
                + "（OnInit 时 data1 未就绪，已解锁开仓）");
        }
        Else {
            WLogE("MAPPING_RECHECK ok=False data0=" + Symbol + " data1=" + D1Sym()
                + " mainC=" + MainContSymbol + " MapToMain=" + IIFString(MapToMain, "True", "False"));
        }
        Return 0;
    }

    // 交易统一门槛：挂图合法（MappingOk）+ 模拟/实盘账户已就绪 + 已到 TradeStartDate + N/四通道五值均 >0。
    // 不通过 = 不走 ManagePosition；实盘/模拟在 data1 失效期另由 ManageStopOnly 保留止损保护。
    Bool CanTradeNow() {
        // 模拟/实盘挂图不合法（非主连/两图层不一致/MapToMain=False）→ 禁止开新仓
        If (MappingOk == False) {
            Return False;
        }
        // 交易模式但账户数据未就绪 → 拒发一切委托（回测无此限制）
        If (IsStrategyTradeMode()) {
            If (IsLiveTrade() == False) {
                Return False;
            }
        }
        If (YMD() < TradeStartDate) {
            Return False;
        }
        If (NVal <= 0 Or EntryLongLv <= 0 Or EntryShortLv <= 0 Or ExitLongLv <= 0 Or ExitShortLv <= 0) {
            Return False;
        }
        Return True;
    }

    // 入场判断：空仓时检查价格是否突破前 EntryN 根 data1 极值（唐奇安通道上/下沿；随最新完成 data1 K线刷新）。
    // 概念映射：多头触发 = High() ≥ EntryLongLv；空头触发 = Low() ≤ EntryShortLv；评估粒度 = 分钟 bar。
    // 系统1 跳过：S1Skip>0 且 SkipNextBreakout 置位（上次同类突破盈利）→ 本次突破取消并消费该标记。
    // 调用前提：仅由 ManagePosition 空仓分支调用（平台空仓、台账已清后）。
    // 互斥/单动作：多头分支 Return 后不再评估空头——同 bar 双突破时多头优先；每 bar 至多开 1 unit。
    // 后置：开仓交给 DoOpenUnit（实际价由 CalcOrderPrice、手数由 CalcLots 解析）；手数不足时逐 bar
    //       重试、OPEN_SKIP 日志按日节流。
    Integer TryEntry() {
        // 参数校验失败：不开新仓，且不消费 SkipNextBreakout / 不写 BREAKOUT_SKIP
        If (ParamsOk == False) {
            Return 0;
        }
        // FixedDir 方向开关（对齐聚宽 fixed_dir）：0=双向 / 1=仅多 / -1=仅空
        Bool allowLong = (FixedDir == 0 Or FixedDir == 1);
        Bool allowShort = (FixedDir == 0 Or FixedDir == -1);
        // 多头突破
        If (allowLong And High() >= EntryLongLv) {
            // 系统1 跳过：消费标记，仅跳过一次突破
            If (S1Skip > 0 And SkipNextBreakout) {
                SkipNextBreakout = False;
                WLog("BREAKOUT_SKIP long level=" + Text(EntryLongLv, 1));
                Return 0;
            }
            // 触发价=通道上沿；实际委托价/手数在 DoOpenUnit 内解析
            DoOpenUnit(1, EntryLongLv, "entry");
            Return 0;
        }
        // 空头突破（多头已触发则上面已 Return——每 bar 单动作）
        If (allowShort And Low() <= EntryShortLv) {
            // 系统1 跳过：消费标记，仅跳过一次突破
            If (S1Skip > 0 And SkipNextBreakout) {
                SkipNextBreakout = False;
                WLog("BREAKOUT_SKIP short level=" + Text(EntryShortLv, 1));
                Return 0;
            }
            // 触发价=通道下沿；实际委托价/手数在 DoOpenUnit 内解析
            DoOpenUnit(-1, EntryShortLv, "entry");
        }
        Return 0;
    }

    // ---------- STALL 极值停滞并行入场（v3.2；口径=data1 周期，默认日线）----------

    // STALL 运行状态复位（OnInit 调用；StallEnabled 随后由 StallPeriod 校验重新置位）
    Integer ResetStallState() {
        StallEnabled = False;
        StallInit = False;
        StallSigO = 0;
        StallSigH = 0;
        StallSigL = 0;
        StallSigC = 0;
        StallLongCnt = -1;
        StallShortCnt = -1;
        StallRefSym = "";
        Return 0;
    }

    // 计数：最新完成 data1 bar 距最近一次“创新低/新高”事件的 bar 数（事件当根=0；无事件=-1）。
    // 事件 := Low(j) < min(Low(j+1..j+n))（多头）/ High(j) > max(High(j+1..j+n))（空头），严格不等。
    // 与聚宽 stall_state 等价：j 对应 i=last-j+1，计数 last-i=j-1；扫描窗口 2n+5 根。
    Integer CalcStallCounts(Integer n) {
        StallLongCnt = -1;
        StallShortCnt = -1;
        Integer lim = n + 5;
        Integer j;
        Integer k;
        Numeric ext;
        For j = 1 To lim {
            If (StallLongCnt < 0) {
                ext = D1Low(j + 1);
                For k = 1 To n {
                    ext = Min(ext, D1Low(j + k));
                }
                If (D1Low(j) < ext) {
                    StallLongCnt = j - 1;
                }
            }
            If (StallShortCnt < 0) {
                ext = D1High(j + 1);
                For k = 1 To n {
                    ext = Max(ext, D1High(j + k));
                }
                If (D1High(j) > ext) {
                    StallShortCnt = j - 1;
                }
            }
        }
        Return 0;
    }

    // STALL 边沿评判与入场。返回 True = 本 bar 已尝试 STALL 开仓（OnBar 跳过常规 ManagePosition）。
    // 边沿 = Data1[1] 的 OHLC 签名变化；每根 data1 bar 只消费一次，首次仅建基线；
    // 持仓/挂单/预热期达成 = 错过，不补触发。换月/换层导致引用键变化时重基线，防复权值跳变误触发。
    // allowOrder=False（历史回放期）：仍推进签名基线与 N/通道，但不下单。
    Bool TryStallEntry(Bool allowOrder) {
        // 参数校验失败：不开新仓（基线推进一并无意义——参数整轮不变）
        If (ParamsOk == False) {
            Return False;
        }
        // StallEnabled 已由 OnInit 统一校验（StallPeriod<=0 或窗口超限 → False）
        If (StallEnabled == False) {
            Return False;
        }
        If (DataCount() < 2) {
            Return False;
        }
        // Data1[1] 四价有效性逐项校验（NumValid：无效值或 <=0 均拒绝；Or 不依赖短路）
        If (NumValid(D1Open(1)) == False Or NumValid(D1High(1)) == False Or
            NumValid(D1Low(1)) == False Or NumValid(D1Close(1)) == False) {
            Return False;
        }
        Numeric so = D1Open(1);
        Numeric sh = D1High(1);
        Numeric sl = D1Low(1);
        Numeric sc = D1Close(1);
        String ref = D1Sym() + "|" + MainSymbol();
        // 首根 / 换月换层：仅建基线，不触发
        If (StallInit == False Or StallRefSym <> ref) {
            StallRefSym = ref;
            StallSigO = so;
            StallSigH = sh;
            StallSigL = sl;
            StallSigC = sc;
            StallInit = True;
            Return False;
        }
        // 同一根 data1 bar 已评判过
        If (so == StallSigO And sh == StallSigH And sl == StallSigL And sc == StallSigC) {
            Return False;
        }
        // 新 data1 bar：先消费边沿，再同步 N/四通道（OnBarOpen 可能仍读到旧 Data1[1]）
        StallSigO = so;
        StallSigH = sh;
        StallSigL = sl;
        StallSigC = sc;
        RefreshNChannels();
        // 回放期：边沿已消费、通道已同步，但不下单（OnBar 已跳过 ManagePosition）
        If (allowOrder == False) {
            Return False;
        }
        If (CanTradeNow() == False) {
            If (LogLevel >= 3) {
                WLogD("STALL_MISS reason=gated");
            }
            Return False;
        }
        CalcStallCounts(StallPeriod);
        If (LogLevel >= 3) {
            WLogD("STALL_EDGE sig=" + Text(so, 1) + "/" + Text(sh, 1) + "/" + Text(sl, 1) + "/" + Text(sc, 1)
                + " long=" + Text(StallLongCnt) + " short=" + Text(StallShortCnt));
        }
        Bool allowLong = (FixedDir == 0 Or FixedDir == 1);
        Bool allowShort = (FixedDir == 0 Or FixedDir == -1);
        Integer side = 0;
        If (allowLong And StallLongCnt == StallPeriod) {   // 多空同达成时长优先（同聚宽）
            side = 1;
        }
        Else {
            If (allowShort And StallShortCnt == StallPeriod) {
                side = -1;
            }
        }
        If (side == 0) {
            Return False;
        }
        String sideTxt = "long";
        If (side < 0) {
            sideTxt = "short";
        }
        If (MarketPosition() <> 0) {
            WLog("STALL_MISS side=" + sideTxt + " reason=holding stall=" + Text(StallPeriod));
            Return False;
        }
        // 已有未完成委托（开/平/加/损）→ 本边沿放弃，不补触发；避免先记 STALL_FIRE 再被门禁拦
        If (GetArraySize(PendingIds) > 0) {
            WLog("STALL_MISS side=" + sideTxt + " reason=pending ids=" + PendingStr());
            Return False;
        }
        // 平台空仓但台账残留：先清再开（与 ManagePosition 空仓分支同口径）
        If (PLots <> 0 Or UnitCount <> 0) {
            WLogW("LEDGER_CLEAN 空仓残留台账清理 prevLots=" + Text(PLots));
            ResetLedger();
        }
        WLog("STALL_FIRE side=" + sideTxt + " stall=" + Text(StallPeriod)
            + " long=" + Text(StallLongCnt) + " short=" + Text(StallShortCnt)
            + " N=" + Text(NVal, 1));
        // 无论手数不足/委托被拒，均视为本 bar 已尝试（对齐聚宽：STALL 分支不再跑海龟检查）
        DoOpenUnitMarket(side, "entry_stall");
        Return True;
    }

    // data0 心跳（仅实时 OnBarClose，每根一行）：证明事件循环在跑，并留下价格/仓位/N 状态。
    // 有 stop 才附 stop，有挂单才附 pend，保持短行。
    Integer LogBarHeartbeat() {
        String stopSeg = "";
        If (MarketPosition() <> 0) {
            stopSeg = " stop=" + Text(StopLine, 1);
        }
        String pendSeg = "";
        If (GetArraySize(PendingIds) > 0) {
            pendSeg = " pend=" + Text(GetArraySize(PendingIds));
        }
        WLog("HB " + Symbol + " C=" + Text(Close(), 1)
            + " N=" + Text(NVal, 2)
            + " pos=" + Text(MarketPosition())
            + " units=" + Text(UnitCount) + " lots=" + Text(PLots) + stopSeg + pendSeg);
        Return 0;
    }

    // data1 完整行（仅实时 OnBarClose，OnBar 之后调用）：新完成的一根 data1 bar 一行，
    // 含 data1 OHLC + N/四通道 + STALL 计数 + 持仓台账。
    // 用单一字符串指纹去重（合约|OHLC）：同一根 data1 bar 只打印一次。
    Integer LogData1Bar() {
        If (DataCount() < 2) {
            Return 0;
        }
        // 四价有效性逐项校验（NumValid：无效值或 <=0 均拒绝；Or 不依赖短路）
        If (NumValid(D1Open(1)) == False Or NumValid(D1High(1)) == False Or
            NumValid(D1Low(1)) == False Or NumValid(D1Close(1)) == False) {
            Return 0;
        }
        Numeric o = D1Open(1);
        Numeric h = D1High(1);
        Numeric l = D1Low(1);
        Numeric c = D1Close(1);
        String sig = D1Sym() + "|" + MainSymbol() + "|" + Text(o, 4) + "|" + Text(h, 4)
            + "|" + Text(l, 4) + "|" + Text(c, 4);
        If (sig == D1LogSig) {
            Return 0;
        }
        D1LogSig = sig;
        WLog("D1BAR " + D1Sym() + "(" + D1Freq() + ") O=" + Text(o, 1)
            + " H=" + Text(h, 1) + " L=" + Text(l, 1) + " C=" + Text(c, 1)
            + " N=" + Text(NVal, 2)
            + " EL=" + Text(EntryLongLv, 1) + " XL=" + Text(ExitLongLv, 1)
            + " ES=" + Text(EntryShortLv, 1) + " XS=" + Text(ExitShortLv, 1)
            + " stallL=" + Text(StallLongCnt) + " stallS=" + Text(StallShortCnt)
            + " units=" + Text(UnitCount) + " lots=" + Text(PLots) + " stop=" + Text(StopLine, 1));
        Return 0;
    }

    // 交易主流程（每分钟）：按持仓状态路由，每 bar 至多执行一个动作。
    // 空仓分支：清残留台账（LEDGER_CLEAN）→ TryEntry 尝试突破入场。
    // 持仓分支：数量对账（失配 → RebuildLedger）→ 按优先级执行：止损 > 反向突破出场 > 加仓，
    //           先保护后进攻；任一动作完成后立即 Return，本 bar 不再做第二个动作。
    // 概念映射（对齐聚宽）：止损 = 共享 2N 线；出场 = ExitN 反向通道；加仓 = 顺向 AddStepN×N 步进；
    //           入场 = EntryN 唐奇安突破（细节见 TryEntry）。
    // 调用前提：OnBar 第 2 步且 CanTradeNow() 通过（MappingOk + TradeStartDate + N/四通道有效）。
    Integer ManagePosition() {
        // 空仓分支：先清残留台账（手工平仓/异常后可能残留），再尝试入场
        If (MarketPosition() == 0) {
            If (PLots <> 0 Or UnitCount <> 0) {
                WLogW("LEDGER_CLEAN 空仓残留台账清理 prevLots=" + Text(PLots));
                ResetLedger();
            }
            TryEntry();
            Return 0;
        }
        // 持仓对账：平台手数 ≠ 台账 → 以平台为准重建（重启/部分成交/换月后）
        If (IntPart(Abs(CurrentContracts())) <> PLots) {
            RebuildLedger();
        }
        // 持仓方向：+1 多 / -1 空（止损/出场/加仓的判向与触发价用）
        Integer dir = 1;
        If (MarketPosition() < 0) {
            dir = -1;
        }
        // 1) 共享止损：多单 Low ≤ StopLine、空单 High ≥ StopLine（2N 距离，加仓后随末 unit 上移）
        If (StopLine > 0) {
            If (dir > 0 And Low() <= StopLine) {
                DoCloseAll(dir, StopLine, "stop");
                Return 0;
            }
            If (dir < 0 And High() >= StopLine) {
                DoCloseAll(dir, StopLine, "stop");
                Return 0;
            }
        }
        // 2) 反向突破出场：多单跌破 ExitLongLv、空单涨破 ExitShortLv（ExitN 通道反向沿）
        If (dir > 0 And Low() <= ExitLongLv) {
            DoCloseAll(dir, ExitLongLv, "exit_breakout");
            Return 0;
        }
        If (dir < 0 And High() >= ExitShortLv) {
            DoCloseAll(dir, ExitShortLv, "exit_breakout");
            Return 0;
        }
        // 3) 加仓：顺向每 AddStepN*N 加 1 unit；UnitCount < MaxUnits 封顶
        If (UnitCount < MaxUnits) {
            If (dir > 0 And High() >= LastUnitPrice + AddStepN * NVal) {
                DoOpenUnit(dir, LastUnitPrice + AddStepN * NVal, "add");
                Return 0;
            }
            If (dir < 0 And Low() <= LastUnitPrice - AddStepN * NVal) {
                DoOpenUnit(dir, LastUnitPrice - AddStepN * NVal, "add");
                Return 0;
            }
        }
        Return 0;
    }

    // data1 失效时的仅止损兜底（只对实盘/模拟启用）：
    // N/四通道归 0 后 CanTradeNow=False，不能走 ManagePosition
    // （重建会把 StopLine 推到末次入场价（NVal=0 按保本线兜底），零通道也会误判出场/加仓）。
    // 只用最后一次有效 StopLine 保护已有持仓；台账缺失（PLots<=0）时不动作。
    // 注意：失效期间若恰逢换月，UpdateMainSymbol 的重建会把 StopLine 推到末次入场价（NVal=0 → 按保本线兜底）。
    Integer ManageStopOnly() {
        If (MarketPosition() == 0 Or PLots <= 0 Or StopLine <= 0) {
            Return 0;
        }
        Integer dir = 1;
        If (MarketPosition() < 0) {
            dir = -1;
        }
        If (dir > 0 And Low() <= StopLine) {
            WLogW("STOP_ONLY data1无效，按最后有效止损线 " + Text(StopLine, 1) + " 平多");
            DoCloseAll(dir, StopLine, "stop");
            Return 0;
        }
        If (dir < 0 And High() >= StopLine) {
            WLogW("STOP_ONLY data1无效，按最后有效止损线 " + Text(StopLine, 1) + " 平空");
            DoCloseAll(dir, StopLine, "stop");
            Return 0;
        }
        Return 0;
    }

                + " data1Same=" + IIFString(layerOk, "True", "False")

                If (layerOk == False) {

        WLogInit("GATE: 挂单门禁=OnOrder 未完成委托集合；回放期门禁=模拟/实盘非实时 bar 不下单；图表模式自动交易开（图表交易）一律不发单；简单回测与图表回测不拦");
        WLogInit("参数A: FixedDir=" + Text(FixedDir) + " NPeriod=" + Text(NPeriod) + " NSmooth=" + Text(NSmooth)
            + " EntryN=" + Text(EntryN) + " ExitN=" + Text(ExitN) + " StallPeriod=" + Text(StallPeriod)
            + " MaxUnits=" + Text(MaxUnits));
        String stallFreq = "na";
        If (DataCount() >= 2) {
            stallFreq = D1Freq();
        }
        WLogInit("STALL: enabled=" + IIFString(StallEnabled, "True", "False")

    // 新 bar 生成时驱动一次（正常历史回放与实盘均驱动；参数 indexs=本次新生成的图层数组，本公式不使用）
    OnBarOpen(ArrayRef<Integer> indexs) {
        // [红线] 同 OnBar：只放 Range[0:0]——策略体只在 data0（1分钟）上下文执行一次（多图层防双跑）。
        Range[0:0] {
            // N/四通道刷新：每根新 bar 一次——实盘形成中 bar 的逐 tick OnBar 驱动不会重复重算。
            // 刻意不做「仅 data1 新 bar 时」过滤：刷新幂等（值随 data1 头部推进才变）、边界读数最迟下根 bar 自纠。
            // 失败（无 data1 源/合约错位/历史不足/参数越界）→ 五值归 0 拒交易，下根新 bar 自动重试；日志按交易日节流。
            RefreshNChannels();
        }
    }

    OnBar(ArrayRef<Integer> indexs) {
        // [红线] 只放 Range[0:0]：策略体只在 data0（1分钟）上下文执行一次。
        // 若写成 Range[0:1]/[0:DataCount-1]，每个图层上下文各跑一遍 → 双跑事故。
        Range[0:0] {
            BarCnt = BarCnt + 1;

            // 首 bar 再清一次挂单集合：OnInit 内的数组操作若未生效，避免上次运行的 orderId 残留致门禁死锁
            If (BarCnt == 1) {
                PendingClear();
            }

            // 实盘防呆：TradeStartDate 未到时 CanTradeNow 恒 False、静默不交易 → 首 bar 警告一次（重启复现）。
            // 放在首 bar 而非 OnInit：YMD() 依赖 bar 上下文，OnInit 里判不了；挂上策略的第一分钟即可见。
            If (BarCnt == 1 And IsLiveTrade() And YMD() < TradeStartDate) {
                WLogW("TRADE_GATE 实盘未到 TradeStartDate=" + Text(TradeStartDate) + "，不会产生交易动作");
            }

            // ---- 1. 主力移仓检测（原生自动换仓）----
            UpdateMainSymbol();
            // 1b. 挂图合法性复核：OnInit 时 data1 可能未就绪，首个数据齐备的 bar 解锁一次
            RecheckMapping();

            // ---- 2. STALL 并行入场（先于常规海龟检查）----
            // 回放期门禁：模拟/实盘非实时 bar 不下单（否则历史 bar 委托被拒→乐观台账→
            // 下一 tick 清理→重试，形成 OPEN_FAIL/信号闪烁循环）。只推进 STALL 基线与通道，
            // 不调用 ManagePosition；QuoteStatus=RealTime 的首根 bar 恢复交易。回测不受影响。
            // 两种不可交易阶段：①模拟/实盘的非实时 bar ②图表模式自动交易开（图表交易）。
            // 账户未就绪另由 CanTradeNow 拦。
            If (IsTradablePhase() == False) {
                If (ReplayGateLogged == False) {
                    ReplayGateLogged = True;
                    WLog("REPLAY_GATE 跳过交易：只更新通道/STALL 基线（mode=" + Text(StrategyMode())
                        + " QuoteStatus=" + Text(QuoteStatus()) + "）");
                }
                TryStallEntry(False);
            }
            Else {
                If (IsStrategyTradeMode()) {
                    If (RealtimeLogged == False) {
                        RealtimeLogged = True;
                        WLog("REALTIME_BAR QuoteStatus=" + Text(QuoteStatus()) + " 恢复交易");
                    }
                }
                // True=边沿上确实尝试了开仓 → 本 bar 跳过常规检查（对齐聚宽 stall 分支的 elif）；
                // False=无边沿/边沿已消费 → 照常走止损/出场/加仓（持仓中达成 STALL 时仍管理持仓，
                // 此点与聚宽 elif 不同，为有意偏差，见说明文档）。
                If (TryStallEntry(True) == False) {
                    // 风控与信号（N/四通道无效期 levels=0 → CanTradeNow=False，不下单）
                    If (CanTradeNow()) {
                        ManagePosition();
                    }
                    Else {
                        // 交易模式但账户数据未就绪：暂停一切委托（CanTradeNow 已拦），每交易日首条告警
                        If (IsStrategyTradeMode()) {
                            If (IsLiveTrade() == False) {
                                If (YMD() >= TradeStartDate) {
                                    If (LastAcctWarnYmd <> YMD()) {
                                        LastAcctWarnYmd = YMD();
                                        WLogW("ACCT_NOT_READY 交易模式但账户数据未就绪，暂停交易（检查账户连接/绑定）");
                                    }
                                }
                            }
                        }
                        // data1 失效期：实盘/模拟仍用最后有效止损线保护持仓；
                        // 回测保持原语义，TradeStartDate 之前也不动作。
                        If (IsLiveTrade() And YMD() >= TradeStartDate) {
                            ManageStopOnly();
                        }
                    }
                }
            }

            // ---- 3. 每日定时日志（14:45，同聚宽 signal_time）----
            // 回测照写；模拟/实盘仅实时（历史回放阶段不写）。
            Bool dailyLogOk = True;
            If (IsStrategyTradeMode()) {
                dailyLogOk = (QuoteStatus() == Enum_QuoteStatus_RealTime);
            }
            If (dailyLogOk And Hour() == LogHour And Minute() == LogMinute And YMD() <> LastLogYmd) {
                LastLogYmd = YMD();
                String d1s = "na";
                String d0seg = "";
                String pendseg = "";
                If (DataCount() >= 2) {
                    d1s = Text(D1Close(1), 1);
                    // d0diag 仅 DEBUG 级：经 D1Close0Diag 读 D0 禁区（回测=当天完整收盘/实盘=形成中）仅供诊断，禁入判断
                    If (LogLevel >= 3) {
                        d0seg = " d0diag=" + Text(D1Close0Diag(), 1);
                    }
                }
                // 挂单门禁状态：仅在存在未完成委托时附加（回测集合恒空 → 日志与 v3.1 一致）
                If (GetArraySize(PendingIds) > 0) {
                    pendseg = " pend=" + Text(GetArraySize(PendingIds)) + " ids=" + PendingStr();
                }
                WLog("DAILY " + MainSymbol()
                    + " N=" + Text(NVal, 1) + " EL=" + Text(EntryLongLv, 1) + " XL=" + Text(ExitLongLv, 1)
                    + " ES=" + Text(EntryShortLv, 1) + " XS=" + Text(ExitShortLv, 1)
                    + " units=" + Text(UnitCount) + " lots=" + Text(PLots) + " stop=" + Text(StopLine, 1)
                    + " equity=" + Text(CurEquity(), 0) + " win=" + Text(WinSize())
                    + " d1=" + d1s + d0seg + pendseg);
            }

            // ---- 4. 结束摘要：回测专用留痕（整段跑完的单行"期末对账"）----
            // 本段主要为回测验收/回归对照而设：不翻全程日志，一行核对三件事——
            //   ① 对账：pos/contracts（平台）vs lots（台账），不等=台账漂移（查 STATE_REBUILD）。
            //   ② 收益：equity 期末权益为主指标；BarCnt=处理的1分钟bar数 → 证明整段跑完。
            //   ③ 体检：DataCount=2（日线图层在）、N 应为合理正值、d1 日线源最新完成收盘。
            // 触发 BarStatus()==2 = 最右末根（0=首根 / 1=中间 / 2=末根）：回测=整段最后一根，只此一次。
            // 只在回测输出：实盘/模拟的最新 bar 天然最右，挂载后首根会命中并把初始快照误标成"运行结束"。
            If (IsBacktestMode()) {
                If (BarStatus() == 2 And FinalDone == False) {
                    FinalDone = True;          // 幂等守卫：只打一次，OnInit 复位
                    // 与 DAILY 同守卫：无日线图层时不碰 Data1（访问非法）→ d1e 保持空、字段自然省略
                    String d1e = "";
                    If (DataCount() >= 2) {
                        d1e = " d1=" + Text(D1Close(1), 1);
                    }
                    WLog("=== 运行结束 BarCnt=" + Text(BarCnt) + " win=" + Text(WinSize()) + " DataCount=" + Text(DataCount())
                        + " N=" + Text(NVal, 1) + " pos=" + Text(MarketPosition()) + " contracts=" + Text(CurrentContracts())
                        + " lots=" + Text(PLots) + " equity=" + Text(CurEquity(), 0) + d1e + " ===");
                }
            }
        }
    }

    // 每根 data0 bar 收盘时驱动一次（下一根 bar 开始前，末根不触发）。
    // 逐 bar 诊断只在模拟/实盘实时行情输出：回测与历史回放不刷日志。
    // indexs=本次驱动收盘的图层集：心跳只在含 0 时打，滤掉 data1 单独收盘的额外事件。
    OnBarClose(ArrayRef<Integer> indexs) {
        // [红线] 同 OnBar：只放 Range[0:0]，避免多图层各跑一遍
        Range[0:0] {
            If (IsStrategyTradeMode()) {
                If (QuoteStatus() == Enum_QuoteStatus_RealTime) {
                    If (ArrayFind(indexs, 0)) {
                        LogBarHeartbeat();
                    }
                    LogData1Bar();
                }
            }
        }
    }

    // ==================== 回报事件 ====================
    // 回测不触发（EventProbe 实测 0 次）；模拟盘实测回调正常（RateCheckProbe 2026-10-19：
    // OnOrder 1(Declare)→2(Declared)→5(Filled)，OnFill 的 side/combOffset/price/fillPrice 有效）；
    // 实盘是否回调仍由柜台决定。
    // OnOrder 维护挂单门禁（PendingIds），不改台账；OnFill 仅日志，不动 PLots/PCost/StopLine。
    // FillRef：volume/fillVolume/orderId 为 Integer，fillId 为 String（勿套 Text）；
    // 后三字段模拟盘填充待实测，仅日志留痕。

    OnFill(FillRef ordFill) {
        FillCnt = FillCnt + 1;
        WLogEvt("FILL #" + Text(FillCnt)
            + " id=" + Text(ordFill.orderId)
            + " fillId=" + ordFill.fillId
            + " side=" + IIFString(ordFill.side == Enum_Buy, "Buy", "Sell")
            + " off=" + IIFString(ordFill.combOffset == Enum_Entry, "Entry", "Exit")
            + " vol=" + Text(ordFill.volume)
            + " fillVol=" + Text(ordFill.fillVolume)
            + " price=" + Text(ordFill.price, 1)
            + " fillPrice=" + Text(ordFill.fillPrice, 1)
            + " lots=" + Text(PLots) + " units=" + Text(UnitCount)
            + " stop=" + Text(StopLine, 1));
    }

    OnOrder(OrderRef ord) {
        OrderCnt = OrderCnt + 1;
        // 挂单门禁维护：活跃状态入集合；其余（Filled/Canceled/Deleted/未知）视为终态移除
        Integer oid = ord.orderId;
        String ptxt = "";
        If (OrderIsPending(ord.status)) {
            If (PendingIndex(oid) < 0) {
                PendingAdd(oid);
                ptxt = " PEND_ADD";
            }
        }
        Else {
            If (PendingIndex(oid) >= 0) {
                PendingRemove(oid);
                ptxt = " PEND_DEL";
            }
        }
        WLogEvt("ORDER #" + Text(OrderCnt)
            + " id=" + Text(ord.orderId)
            + " status=" + Text(ord.status)
            + " side=" + IIFString(ord.side == Enum_Buy, "Buy", "Sell")
            + " off=" + IIFString(ord.combOffset == Enum_Entry, "Entry", "Exit")
            + " vol=" + Text(ord.volume)
            + " fvol=" + Text(ord.fillVolume)
            + " px=" + Text(ord.price, 1)
            + " lots=" + Text(PLots) + " units=" + Text(UnitCount)
            + " pending=" + Text(GetArraySize(PendingIds)) + ptxt);
    }

    // 停止策略单元时汇总事件到达情况（回测不输出，保持回归日志不变）
    OnExit() {
        If (StrategyMode() == Enum_Mode_Strategy_Trade()) {
            WLogEvt("EXIT fill=" + Text(FillCnt) + " order=" + Text(OrderCnt)
                + " pending=" + Text(GetArraySize(PendingIds)) + " ids=" + PendingStr());
        }
    }
