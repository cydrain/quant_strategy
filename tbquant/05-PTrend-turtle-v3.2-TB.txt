// 简称: PTrendTurtle_v32
// 名称: 棕榈油海龟趋势策略 v3.2（TBQuant · 双图层 + STALL / NR 并行入场）
// 类别: 趋势跟踪
// 类型: 用户应用

// 指标与 STALL/NR 口径 = data1 图层周期（默认日线；30m 等日内周期为实验口径，与聚宽不可比）。
// 窗口头为最新一根已完成 data1 K线，OnBarOpen 每根新 bar 刷新一次；全局状态随 OnInit 复位。
//
// STALL（StallPeriod>0）：data1 创此前 StallPeriod 根新低/新高后，最新完成 bar 距该事件
//   恰 StallPeriod 根 → 下一根 1m 市价开 1 unit。空仓才触发；FixedDir=0 多空同达成则放弃；
//   每根 data1 bar 只评判一次，持仓/挂单/预热期达成 = 错过不补。
//   触发时点与聚宽一致：data1 推进后的首根 1m（日线通常夜盘 21:01）。
// NR（NrN>=2，默认 6）：最新完成 bar 为近 NrN 根振幅最小者 → 下一根 data1 bar 内 1m
//   触及该 bar 高/低点入场。优先级 STALL > NR > EntryN 突破；每事件仅一次，
//   开仓失败下一分钟重试；事件在下一根 data1 bar 内有效（1d 从夜盘 21:01 起）。
// 系统1 跳过（S1Skip>0）：S1Scope='breakout'（默认）仅 20 日突破首仓盈利武装并消费；
//   'breakout_nr' 突破/NR 各自独立；'current' 任意盈利首仓都武装突破+NR。STALL 不参与。
//
// 挂载（两图层缺一不可；K线收盘模型）：
//   data0=1分钟（信号/执行）；data1=同合约日线（只读指标源）。
//   MapToMain=True：主连图（后复权/映射真实价/自动换仓/委托映射主力）；
//   单合约切片回测必须改 False。模拟/实盘要求 data0=data1=MainContSymbol 且 MapToMain=True。
//   模拟/实盘挂单未完成时禁发新单；非实时 bar（历史回放）不下单；图表自动交易模式一律不发单。
//   模拟/实盘下单走 A_* 账户接口（落地合约+真实价）；回测走 Buy/SellShort/Sell/BuyToCover。
//
// 读法红线（改代码勿破）：
//   1) N=Data1[1..WinSize]，第 WinSize 根仅作前收盘锚点；通道=Data1[1..EntryN/ExitN]。
//   2) 信号/风控只读 Data1[1] 及更早；Data1[0] 是禁区（回测=look-ahead，仅 DEBUG d0diag 例外）。
//   3) 禁 GetBar；Data1[1] 恒为最新完成K。
//   4) 信号体只放 Range[0:0]，Range[0:1] 会每图层各跑一遍。
//   5) Data1 仅出现在「数据源约定」访问器段。
//
// 预热：首个目标信号日前 data1 须已有 >=61 根日线（WinSize=61，约 3 个月）；不足时
//   窗口越界读数可能仍是有效值 → N 偏高/假突破。两图层数据范围要同步提前。
//------------------------------------------------------------------------

    // ---------- 基础辅助 ----------

    // 日志目录规范化：非空且末位不是 / 时自动补 /
    String EffectiveLogDir() {
        If (LogDir == "") {
            Return "";
        }
        If (Right(LogDir, 1) == "/") {
            Return LogDir;
        }
        Return LogDir + "/";
    }

    // 回测模式 = 既非交易也非图表（简单回测实测 StrategyMode=4；图表模式=0 另算）。
    Bool IsBacktestMode() {
        Return StrategyMode() <> Enum_Mode_Strategy_Trade() And StrategyMode() <> Enum_Mode_Strategy_Chart();
    }

    // 日志路径：目录 + 公式名 + data0 合约 + [非交易 _BT] + 平台日期
    String LogPath() {
        String modeTag = "";
        If (StrategyMode() <> Enum_Mode_Strategy_Trade()) {
            modeTag = "_BT";
        }
        Return EffectiveLogDir() + FormulaName() + "_" + Symbol + modeTag + "_日志_" + Text(CurrentDate());
    }

    // 时间戳：WLog*（bar 上下文）用 bar 日期时间；OnInit 阶段无 bar 时间 → 只能用 WLogInit*，取平台当前时间
    String BarStamp() {
        Return Text(Date()) + " " + Text(Time());
    }

    String InitStamp() {
        Return DateTimeToString(CurrentDate());
    }

    // 日志核心：LogLevel 阈值过滤；文件 +（PrintToUI 时）图表控制台双写
    Integer WLogCore(Integer need, String tag, String ts, String s) {
        If (LogLevel >= need) {
            String msg = ts + " | " + tag + " " + s;
            FileAppend(LogPath(), msg, False);
            If (PrintToUI) {
                Print(msg);
            }
        }
        Return 0;
    }

    Integer WLogE(String s) {
        Return WLogCore(0, "!!! [Error] !!!", BarStamp(), s);
    }

    Integer WLogW(String s) {
        Return WLogCore(1, "[Warn]", BarStamp(), s);
    }

    Integer WLog(String s) {
        Return WLogCore(2, "[Info]", BarStamp(), s);
    }

    Integer WLogD(String s) {
        Return WLogCore(3, "[Debug]", BarStamp(), s);
    }

    Integer WLogInitE(String s) {
        Return WLogCore(0, "!!! [Error] !!!", InitStamp(), s);
    }

    Integer WLogInit(String s) {
        Return WLogCore(2, "[Info]", InitStamp(), s);
    }

    // 回报事件日志：事件在 bar 上下文外驱动，时间戳取平台时钟（Date()/Time() 可能为旧值）
    Integer WLogEvt(String s) {
        Return WLogCore(2, "[Event]", DateTimeToString(SystemDateTime()), s);
    }

    Integer YMD() {
        Return Year() * 10000 + Month() * 100 + Day();
    }

    // 授权有效期（公式内常量，编译进 .nef；不暴露为 Params）
    Integer LicenseEndYmd() {
        Return 20261231;
    }

    // 回测/图表模式不校验；模拟/实盘按机器日期 CurrentDate()（非K线日期）判断。
    // 仅劝退：.nef 可反编译、系统时钟可回拨，不能替代服务端/原生授权。
    Bool LicenseValid() {
        If (StrategyMode() <> Enum_Mode_Strategy_Trade()) {
            Return True;
        }
        Return IntPart(CurrentDate()) <= LicenseEndYmd();
    }

    // 数值有效性：InvalidNumeric（≈1.79e308）或 <=0 均为无效。
    // 成对合取，不依赖 And 短路；用于 OHLC、N、委托价等逐值校验。
    Bool NumValid(Numeric value) {
        Return (value != InvalidNumeric And value > 0);
    }

    // 刷新失败日志（按交易日节流）：失败态每根新 bar 都会重试——每天只报首条防刷屏
    Integer WLogRefreshErr(String s) {
        If (YMD() <> LastRefreshErrYmd) {
            LastRefreshErrYmd = YMD();
            WLogE(s);
        }
        Return 0;
    }

            ParamErr("AddStepN", Text(AddStepN, 2), ">0");

        }

            ParamErr("StopDistN", Text(StopDistN, 2), ">0");

    // ---------- 数据源约定：全公式仅有的 Data1 引用点 ----------

    // 日线第 idx 根的真实波幅（idx>=1；用 idx 与 idx+1 两根，后者作前收盘锚点）；禁 GetBar。
    Numeric D1TR(Integer idx) {
        Numeric t1 = D1High(idx) - D1Low(idx);
        Numeric t2 = Abs(D1High(idx) - D1Close(idx + 1));
        Numeric t3 = Abs(D1Low(idx) - D1Close(idx + 1));
        Return Max(Max(t1, t2), t3);
    }

    // N：经典海龟口径（同聚宽 calc_n）：前 NPeriod 个 TR 均值作种子，再按
    // N=((NSmooth-1)*N+TR)/NSmooth 递推。窗口 Data1[1..win]，第 win 根仅作前收盘锚点。
    Numeric CalcD1N(Integer win) {
        Integer t;
        Numeric n = 0;
        For t = 1 To NPeriod {
            n = n + D1TR(win - t);
        }
        n = n / NPeriod;
        For t = NPeriod + 1 To win - 1 {
            n = ((NSmooth - 1) * n + D1TR(win - t)) / NSmooth;
        }
        // NaN 守卫：参数滥用产生除零 → 归一为 0，由 RefreshNChannels 记 ERROR 并拒交易
        If (NumValid(n)) {
            Return n;
        }
        Return 0;
    }

    // 四通道（唐奇安）：Data1[1]=最近一根完整日K，其后按各自周期数取极值。
    // 写入 EntryLongLv/EntryShortLv/ExitLongLv/ExitShortLv；调用前须通过 RefreshNChannels 守卫。
    Integer CalcChannels() {
        Numeric eh = D1High(1);
        Numeric el = D1Low(1);
        Numeric xl = D1Low(1);
        Numeric xs = D1High(1);
        Integer j;
        For j = 2 To EntryN {
            eh = Max(eh, D1High(j));
            el = Min(el, D1Low(j));
        }
        For j = 2 To ExitN {
            xl = Min(xl, D1Low(j));
            xs = Max(xs, D1High(j));
        }
        EntryLongLv = eh;
        EntryShortLv = el;
        ExitLongLv = xl;
        ExitShortLv = xs;
        Return 0;
    }

    // N 与四通道复位为无效值（0）：刷新「先清后算」与 OnInit 复位共用
    Integer ResetChannels() {
        NVal = 0;
        EntryLongLv = 0;
        EntryShortLv = 0;
        ExitLongLv = 0;
        ExitShortLv = 0;
        Return 0;
    }

    // N/四通道刷新：先清零，再依次检查 CalcOk、data1 存在且同合约、历史深度，
    // 通过后调用 CalcD1N/CalcChannels；任何失败保持五值 0 并拒交易。
    // 调用时机：OnBarOpen 每根新 bar 一次；TryStallEntry 消费新 data1 边沿时补刷一次。
    Integer RefreshNChannels() {
        Integer win = WinSize();
        ResetChannels();
        // 计算口径参数非法（OnInit 已记 ERROR）：五值保持 0，避免带病计算
        If (CalcOk == False) {
            Return 0;
        }
        If (DataCount() < 2) {
            WLogRefreshErr("N/四通道刷新失败 NO Data1 source, DataCount=" + Text(DataCount()) + "（需追加同合约 data1 图层）");
            Return 0;
        }
        If (D1Sym() <> Symbol) {
            WLogRefreshErr("N/四通道刷新失败 data1.Symbol≠data0.Symbol, data1=" + D1Sym() + " data0=" + Symbol + "（图层顺序/合约不符，检查图表数据源）");
            Return 0;
        }
        // 历史深度守卫：根数不足时序列读取返回无效值（官方：数据不足→无效值）。
        // NumValid 同时排除 InvalidNumeric 与 <=0，不依赖 And 短路。
        Bool valid = NumValid(D1Close(1)) And NumValid(D1Close(win));
        If (valid == False) {
            WLogRefreshErr("N/四通道刷新失败 D1 history too short, win=" + Text(win)
                + " Close[1]=" + IIFString(D1Close(1) != InvalidNumeric, "ok", "invalid")
                + " Close[win]=" + IIFString(D1Close(win) != InvalidNumeric, "ok", "invalid")
                + "（data1 历史不足或缺失）");
            Return 0;
        }
        // 前置守卫已通过 → 窗口内各 bar 有效，交计算函数
        NVal = CalcD1N(win);
        If (NVal <= 0) {   // CalcD1N 已做 NaN 守卫（参数滥用归 0）；此处保持「失败=全 0 + ERROR」语义，CalcChannels 不执行
            WLogRefreshErr("N/四通道刷新失败 CalcD1N 无效, NVal=" + Text(NVal, 2) + "（检查 NPeriod/NSmooth 参数）");
            Return 0;
        }
        CalcChannels();
        Return 0;
    }

    // ---------- 资金分支：回测用 Portfolio_*，实盘/模拟用 A_* ----------

    // 策略交易模式（模拟/实盘图表）：不等同于账户已就绪，账户数据可能晚于 OnInit 才就绪。
    Bool IsStrategyTradeMode() {
        Return StrategyMode() == Enum_Mode_Strategy_Trade();
    }

    // 实盘/模拟判定：策略交易模式 + 账户数据已就绪（回测 A_AccountDataExist=False）
    Bool IsLiveTrade() {
        If (IsStrategyTradeMode() == False) {
            Return False;
        }
        Return A_AccountDataExist(0);
    }

    // ---------- 持仓口径：回测=图表函数；模拟/实盘=台账（由账户真值对账维护）----------
    // 图表口径（MarketPosition/CurrentContracts/AvgEntryPrice）看不到「委托映射主力」后的真实持仓，
    // 模拟/实盘一律改用台账 + 4 参 A_GetPosition 对账；回测沿用图表函数（历史回测口径不变）。
    Integer PosDirNow() {
        If (IsLiveTrade()) {
            Return PDir;
        }
        If (MarketPosition() > 0) {
            Return 1;
        }
        If (MarketPosition() < 0) {
            Return -1;
        }
        Return 0;
    }

    Integer PosLotsNow() {
        If (IsLiveTrade()) {
            Return PLots;
        }
        Return IntPart(Abs(CurrentContracts()));
    }

    Numeric PosAvgChart() {
        If (IsLiveTrade()) {
            If (PLots > 0) {
                Return PCost / PLots;
            }
            Return 0;
        }
        Numeric a = AvgEntryPrice();
        If (a <= 0 And PLots > 0) {
            Return PCost / PLots;
        }
        Return a;
    }

    // 当前阶段是否允许发单（True=可交易）。只管「阶段」；账户/日期/通道等业务条件由 CanTradeNow 判断。
    // 简单回测/参数优化(mode≠0,1)=全部 bar 可交易；模拟/实盘(mode=1)=仅实时 bar；
    // 图表模式(mode=0)=自动交易关（图表回测）可交易；自动交易开（图表交易）一律不发单。
    Bool IsTradablePhase() {
        Bool inBacktest = IsBacktestMode();      // 简单回测/参数优化：mode≠Trade 且 ≠Chart
        Bool inTrade = IsStrategyTradeMode();    // 模拟/实盘策略交易：mode=Trade
        Bool isRealtimeBar = (QuoteStatus() == Enum_QuoteStatus_RealTime);

        // 1) 简单回测/参数优化（mode≠0,1）：每根 bar 都参与模拟成交
        If (inBacktest) {
            Return True;
        }
        // 2) 模拟/实盘（mode=1）：只允许当前实时 bar
        If (inTrade) {
            Return isRealtimeBar;
        }
        // 3) 图表模式（mode=0）：自动交易关=图表回测→可交易；自动交易开=图表交易→一律不发单
        Return IsTradeEnabled() == False;
    }

    // 总权益（含浮盈）：实盘→A_CurrentEquity(0)；回测→Portfolio_CurrentEquity()。供权益日志。
    Numeric CurEquity() {
        If (IsLiveTrade()) {
            Return A_CurrentEquity(0);
        }
        Return Portfolio_CurrentEquity();
    }

    // 风险资金：Min(InitCapital, 当前权益)，回测/实盘同构——盈利超过参数不放大手数，
    // 亏损时按权益缩小。供 CalcLots 风险段与占用上限。
    Numeric RiskCapital() {
        If (IsLiveTrade()) {
            Return Min(InitCapital, A_CurrentEquity(0));
        }
        Return Min(InitCapital, Portfolio_CurrentEquity());
    }

    // 可用资金：实盘→A_FreeMargin(0)（空闲保证金）；回测→Portfolio_CurrentCapital()。
    // 供 CalcLots 资金上限。
    Numeric CurAvailable() {
        If (IsLiveTrade()) {
            Return A_FreeMargin(0);
        }
        Return Portfolio_CurrentCapital();
    }

    // ---------- 手数计算（复刻聚宽 unit_lots + fit_margin + fit_cap）----------

    // 开仓手数：风险定量 → 可用资金 → 占用上限，三段逐级取小。
    // 模拟/实盘按 Rollover 把图价/N 折回真实口径；回测 r=1。
    Integer CalcLots(Numeric price) {
        Numeric eq = RiskCapital();
        Numeric roll = 1;
        If (IsLiveTrade()) {
            Numeric rr = Rollover();
            If (NumValid(rr) And rr > 0) {
                roll = rr;
            }
        }
        // ① 经典海龟 unit：1N 波动 ≈ 风险资金 × UnitRisk（1%），向下取整、最低 1 手
        Integer lots = Max(1, IntPart(eq * UnitRisk * roll / (NVal * MultiplierP)));
        Numeric marginPerLot = price * MultiplierP * MarginRateP / roll;   // 每手保证金（真实价口径）
        If (marginPerLot > 0) {   // ≤0 时跳过②③（防除零；聚宽同款守卫）
            // ② 物理约束：可用资金 ÷ 每手保证金（聚宽 fit_lots_to_margin）
            lots = Min(lots, IntPart(CurAvailable() / marginPerLot));
            If (MarginCapPct > 0) {   // =0 表示不限
                // ③ 占用上限：剩余额度 = 风险资金 × MarginCapPct − 已持仓占用（聚宽 fit_lots_to_cap）
                Numeric room = eq * MarginCapPct - PLots * marginPerLot;
                lots = Min(lots, IntPart(room / marginPerLot));
            }
        }
        Return IIF(lots < 0, 0, lots);   // ②③ 可削到负 → 归 0，语义 = 开不出
    }

    // ---------- 订单封装 ----------

    // 委托价：实盘取对手价（盘口无效退回 refPrice）；回测按触发价，跳空时按开盘价。
    Numeric CalcOrderPrice(Integer orderDir, Numeric refPrice) {
        If (IsLiveTrade()) {
            Numeric qp = Q_BidPrice(0);   // 卖方向默认取买一
            If (orderDir > 0) {
                qp = Q_AskPrice(0);   // 买方向改取卖一（对手价，跨价差保成交）
            }
            If (NumValid(qp)) {   // 盘口无效时 Q_* 返回 InvalidNumeric（≈1.79e308）或 0
                Return qp;
            }
            Return refPrice;   // 盘口无效（未连线/休市/历史回放）→ 理论价兜底
        }
        If (orderDir > 0 And Open() > refPrice) {   // 高开越过触发价：买单按开盘价（对买方更贵）
            Return Open();
        }
        If (orderDir < 0 And Open() < refPrice) {   // 低开越过触发价：卖单按开盘价（对卖方更低）
            Return Open();
        }
        Return refPrice;   // 无跳空：按触发价
    }

    // ---------- 挂单门禁（模拟/实盘）----------
    // OnOrder 维护未完成委托集合；状态枚举扩展时未知状态按终态移除，避免死锁。
    Bool OrderIsPending(Integer st) {
        Return (st == Enum_Declare Or st == Enum_Declared Or st == Enum_FillPart Or st == Enum_Canceling);
    }

    Integer PendingIndex(Integer oid) {
        Integer n = GetArraySize(PendingIds);
        Integer i;
        For i = 0 To n - 1 {
            If (IntPart(PendingIds[i]) == oid) {
                Return i;
            }
        }
        Return -1;
    }

    Integer PendingAdd(Integer oid) {
        If (PendingIndex(oid) < 0) {
            ArrayPushBack(PendingIds, oid);
        }
        Return 0;
    }

    Integer PendingRemove(Integer oid) {
        Integer idx = PendingIndex(oid);
        If (idx >= 0) {
            ArrayErase(PendingIds, idx, 1);
        }
        Return 0;
    }

    Integer PendingClear() {
        Integer n = GetArraySize(PendingIds);
        Integer i;
        For i = 1 To n {
            ArrayErase(PendingIds, 0, 1);
        }
        Return 0;
    }

    String PendingStr() {
        String s = "";
        Integer n = GetArraySize(PendingIds);
        Integer i;
        For i = 0 To n - 1 {
            If (s <> "") {
                s = s + ",";
            }
            s = s + Text(IntPart(PendingIds[i]));
        }
        Return s;
    }

    // 挂单同步用的委托落地合约：优先平台层映射表，其次 MainSymbol，最后图表合约。
    String OrderMapSym() {
        ArrayClear(SyncMapSyms);
        If (GetLayerOrderMapSymbols(SyncMapSyms)) {
            If (GetArraySize(SyncMapSyms) > 0) {
                Return SyncMapSyms[0];
            }
        }
        If (MainSymbol() <> "") {
            Return MainSymbol();
        }
        Return Symbol;
    }

    // 每根 bar 按平台未成交委托列表校正挂单集合：OnOrder 漏报终态时下一根 bar 自动解锁；
    // 查询失败保持原集合不动（防误放行）。仅模拟/实盘执行，回测/图表不调用。
    Integer SyncPendingFromAccount() {
        If (IsLiveTrade() == False) {
            Return 0;
        }
        String sym = OrderMapSym();
        Array<Integer> unfilled;
        Bool ok = A_GetUnFillOrderIDs(sym, unfilled);
        If (ok == False) {
            If (LastSyncWarnYmd <> YMD()) {
                LastSyncWarnYmd = YMD();
                WLogW("PEND_SYNC_FAIL 查询未成交委托失败 sym=" + sym
                    + "，沿用本地集合=" + Text(GetArraySize(PendingIds)) + " ids=" + PendingStr());
            }
            Return 0;
        }
        String before = PendingStr();
        PendingClear();
        Integer n = GetArraySize(unfilled);
        Integer i;
        For i = 0 To n - 1 {
            PendingAdd(unfilled[i]);
        }
        String after = PendingStr();
        If (before <> after) {
            WLog("PEND_SYNC sym=" + sym + " before=[" + before + "] after=[" + after + "]");
        }
        Return 0;
    }

    // ---------- 账户真值（模拟/实盘）：落地合约 + 真实价 + 4 参持仓 ----------

    // 落地真实合约最小变动价位（价格值）。参数 RealTickSize 优先（按合约手填：p=1、ru=5、ag=1）；
    // 主连图上 MinMove 是连续合约口径（p9888 实测 MinMove=1、PriceScale=1），不能直接当真实跳动。
    Numeric RealTick() {
        If (RealTickSize > 0) {
            Return RealTickSize;
        }
        Numeric t = MinMove();
        If (t <= 0) {
            t = MinMove() * PriceScale();
        }
        If (t <= 0) {
            t = 1;
        }
        Return t;
    }

    // 图价 → 真实价，并按委托方向取整（买向上/卖向下）。真实价 = 图价 / Rollover()。
    Numeric RealOrderPrice(Numeric chartPx, Integer orderDir) {
        If (NumValid(chartPx) == False) {
            Return InvalidNumeric;
        }
        Numeric r = Rollover();
        If (NumValid(r) == False) {
            Return InvalidNumeric;
        }
        Numeric rp = chartPx / r;
        If (NumValid(rp) == False) {
            Return InvalidNumeric;
        }
        Numeric t = RealTick();
        Numeric q = rp / t;
        Numeric n = IntPart(q);
        // 浮点误差容差 1e-6 跳：恰好落在跳动上不额外跨价
        If (q - n > 0.999999) {
            n = n + 1;
        }
        If (orderDir > 0) {
            If (q - n > 0.000001) {
                n = n + 1;
            }
        }
        Numeric px = n * t;
        // 消浮点尾差（用乘法缩放，避免整数/整数除法语义差异）
        px = IntPart(px * 1000000 + 0.5) * 0.000001;
        If (px <= 0) {
            Return InvalidNumeric;
        }
        Return px;
    }

    // 读取落地合约账户持仓（4 参 A_GetPosition）。读取失败时用持仓清单区分真空仓与失败。
    // 账户为策略专用；其它策略/手工仓会被一并接管。
    Integer RefreshAccountPosition() {
        String sym = OrderMapSym();
        AcctSym = sym;
        Position p;
        Bool ok = A_GetPosition(sym, p, "", 0);
        AcctPosOk = False;
        If (ok) {
            If (p.longCurrentVolume != InvalidNumeric And p.shortCurrentVolume != InvalidNumeric) {
                AcctLong = p.longCurrentVolume;
                AcctShort = p.shortCurrentVolume;
                AcctLongAvg = p.longAvgPriceO;
                AcctShortAvg = p.shortAvgPriceO;
                AcctPosOk = True;
            }
        }
        Else {
            ArrayClear(AcctSyms);
            If (A_GetPositionSymbols(AcctSyms, 0)) {
                Bool found = False;
                Integer i;
                For i = 0 To GetArraySize(AcctSyms) - 1 {
                    If (AcctSyms[i] == sym) {
                        found = True;
                    }
                }
                If (found == False) {
                    AcctLong = 0;
                    AcctShort = 0;
                    AcctLongAvg = 0;
                    AcctShortAvg = 0;
                    AcctPosOk = True;
                }
            }
        }
        If (AcctPosOk == False) {
            If (LastAcctPosWarnYmd <> YMD()) {
                LastAcctPosWarnYmd = YMD();
                WLogW("ACCT_POS_FAIL 账户持仓读取失败 sym=" + sym + "，暂停开仓（平仓按台账兜底）");
            }
        }
        Return 0;
    }

    // 以账户真值整体重建台账（模拟/实盘唯一重建入口）。均价按 Rollover 折回图口径；
    // 加仓次数/末次入场价无法恢复，止损线按均价重推。不支持同合约多空对锁。
    Integer RebuildLedgerFromAcct() {
        RefreshAccountPosition();
        If (AcctPosOk == False) {
            Return 0;
        }
        Integer al = IntPart(AcctLong);
        Integer as = IntPart(AcctShort);
        If (al > 0 And as > 0) {
            WLogE("STATE_REBUILD src=acct 对锁持仓 long=" + Text(al) + " short=" + Text(as)
                + "（不支持锁仓，暂停交易，需人工处理）");
            AcctPosOk = False;
            Return 0;
        }
        If (al <= 0 And as <= 0) {
            If (PLots <> 0 Or UnitCount <> 0) {
                WLogW("STATE_REBUILD src=acct flat prevLots=" + Text(PLots) + " prevUnits=" + Text(UnitCount));
            }
            PLots = 0;
            PCost = 0;
            UnitCount = 0;
            LastUnitPrice = 0;
            StopLine = 0;
            PDir = 0;
            TradeReason = "";   // 账户已空仓：本笔交易结束，入场类型不可知
            Return 0;
        }
        Integer dir = 1;
        Integer cc = al;
        Numeric avgReal = AcctLongAvg;
        If (as > 0) {
            dir = -1;
            cc = as;
            avgReal = AcctShortAvg;
        }
        Numeric avgChart = 0;
        Numeric r = Rollover();
        If (NumValid(avgReal) And NumValid(r)) {
            avgChart = avgReal * r;
        }
        If (NumValid(avgChart) == False) {
            avgChart = Close();
        }
        PLots = cc;
        PDir = dir;
        If (UnitCount < 1) {
            UnitCount = 1;
        }
        PCost = avgChart * cc;
        LastUnitPrice = avgChart;
        If (NVal > 0) {
            StopLine = LastUnitPrice - dir * StopDistN * NVal;
        }
        Else {
            StopLine = 0;
        }
        WLogW("STATE_REBUILD src=acct dir=" + Text(dir) + " lots=" + Text(cc)
            + " units=" + Text(UnitCount) + " avgReal=" + Text(avgReal, 1)
            + " avgChart=" + Text(avgChart, 1) + " stop=" + Text(StopLine, 1)
            + " N=" + Text(NVal, 2));
        Return 0;
    }

    // 账户接口下单（模拟/实盘唯一出口）：查询账户持仓 → 发送前门禁 → 真实价委托。
    // 平仓手数收敛为实际同向持仓；加仓须已有同向持仓；开仓须账户空仓。
    Bool SendAcctOrder(Integer orderDir, Integer lots, Numeric chartPx, Bool isClose, String reason) {
        String sym = OrderMapSym();
        RefreshAccountPosition();
        Integer lots2 = lots;
        If (isClose) {
            Numeric side = AcctLong;
            If (orderDir > 0) {
                side = AcctShort;   // 买平 → 可平的是空头
            }
            If (AcctPosOk == False) {
                side = PLots;       // 账户不可读：按台账兜底（止损优先于门禁）
            }
            lots2 = IntPart(side);
            If (lots2 <= 0) {
                If (LastCloseSkipYmd <> YMD()) {
                    LastCloseSkipYmd = YMD();
                    WLogW("ORDER_SKIP 无可平持仓 reason=" + reason + " dir=" + Text(orderDir)
                        + " ledgerLots=" + Text(PLots) + " acctOk=" + IIFString(AcctPosOk, "True", "False"));
                }
                Return False;
            }
            If (lots2 <> lots) {
                WLog("CLOSE_CLAMP reason=" + reason + " lots=" + Text(lots) + " -> " + Text(lots2)
                    + " acctLong=" + Text(IntPart(AcctLong)) + " acctShort=" + Text(IntPart(AcctShort))
                    + " acctOk=" + IIFString(AcctPosOk, "True", "False"));
            }
        }
        Else {
            If (reason == "add") {
                Bool sameSide = (orderDir > 0 And AcctLong > 0) Or (orderDir < 0 And AcctShort > 0);
                If (AcctPosOk == False Or sameSide == False) {
                    WLogW("OPEN_BLOCK 加仓需账户已有同向持仓 reason=" + reason + " dir=" + Text(orderDir)
                        + " acctOk=" + IIFString(AcctPosOk, "True", "False")
                        + " acctLong=" + Text(IntPart(AcctLong)) + " acctShort=" + Text(IntPart(AcctShort)));
                    Return False;
                }
            }
            Else {
                If (AcctPosOk == False) {
                    WLogW("OPEN_BLOCK 账户持仓不可读，禁止开仓 reason=" + reason);
                    Return False;
                }
                If (AcctLong > 0 Or AcctShort > 0) {
                    WLogW("OPEN_BLOCK 账户已有持仓，禁止开仓 reason=" + reason
                        + " acctLong=" + Text(IntPart(AcctLong)) + " acctShort=" + Text(IntPart(AcctShort)));
                    RebuildLedgerFromAcct();   // 立即对齐，下一 bar 按持仓管理
                    Return False;
                }
            }
        }
        Numeric px = RealOrderPrice(chartPx, orderDir);
        If (px == InvalidNumeric Or px <= 0) {
            If (LastBadPxYmd != YMD()) {
                LastBadPxYmd = YMD();
                WLogW("ORDER_SKIP 真实委托价无效 reason=" + reason + " pxChart=" + Text(chartPx, 1)
                    + " rollover=" + Text(Rollover(), 4));
            }
            Return False;
        }
        Array<Integer> ids;
        Bool ok = False;
        If (isClose) {
            If (orderDir > 0) {
                ok = A_BuyToCover(sym, lots2, px, ids);
            }
            Else {
                ok = A_Sell(sym, lots2, px, ids);
            }
        }
        Else {
            If (orderDir > 0) {
                ok = A_Buy(sym, lots2, px, ids);
            }
            Else {
                ok = A_SellShort(sym, lots2, px, ids);
            }
        }
        Integer idCnt = GetArraySize(ids);
        Integer i;
        For i = 0 To idCnt - 1 {
            PendingAdd(ids[i]);
        }
        // 平台未返回委托号：挂单门禁无法登记，成交也可能晚于对账 → 延后对账 2 根防台账先归零再重建
        If (ok And idCnt == 0) {
            SyncBlockBars = Max(SyncBlockBars, 2);
            WLogW("ORDER_NO_ID via=acct reason=" + reason + " sym=" + sym
                + " dir=" + Text(orderDir) + " lots=" + Text(lots2) + "（无委托号，延后对账 2 根）");
        }
        LastSentLots = lots2;
        WLog("ORDER_SEND via=acct reason=" + reason + " sym=" + sym
            + " dir=" + Text(orderDir) + " close=" + IIFString(isClose, "True", "False")
            + " lots=" + Text(lots2) + " pxReal=" + Text(px, 1) + " pxChart=" + Text(chartPx, 1)
            + " rollover=" + Text(Rollover(), 4) + " ids=" + Text(idCnt)
            + " ok=" + IIFString(ok, "True", "False"));
        If (ok == False) {
            WLogE("ORDER_REJECT via=acct reason=" + reason + " sym=" + sym
                + " dir=" + Text(orderDir) + " lots=" + Text(lots2) + " pxReal=" + Text(px, 1));
        }
        Return ok;
    }

    // 每根新 bar 按账户真值校正台账；有挂单或宽限期内跳过。
    Integer SyncAcctOnNewBar() {
        If (IsLiveTrade() == False) {
            Return 0;
        }
        String sig = Text(Date()) + "|" + Text(Time());
        If (sig == LastSyncBarSig) {
            Return 0;
        }
        LastSyncBarSig = sig;
        If (GetArraySize(PendingIds) > 0) {
            WLogD("ACCT_SYNC skip pending=" + Text(GetArraySize(PendingIds)));
            Return 0;
        }
        If (SyncBlockBars > 0) {
            SyncBlockBars = SyncBlockBars - 1;
            WLogD("ACCT_SYNC skip grace remain=" + Text(SyncBlockBars));
            Return 0;
        }
        RefreshAccountPosition();
        If (AcctPosOk == False) {
            Return 0;
        }
        Integer al = IntPart(AcctLong);
        Integer as = IntPart(AcctShort);
        If (al > 0 And as > 0) {
            WLogE("ACCT_SYNC 同合约多空同时持仓 long=" + Text(al) + " short=" + Text(as)
                + "（不支持锁仓，暂停交易，需人工处理）");
            AcctPosOk = False;
            Return 0;
        }
        Integer acctLots = al + as;
        Integer acctDir = 0;
        If (al > 0) {
            acctDir = 1;
        }
        If (as > 0) {
            acctDir = -1;
        }
        If (PLots <> acctLots Or PDir <> acctDir Or (acctLots > 0 And StopLine <= 0 And NVal > 0)) {
            RebuildLedgerFromAcct();
        }
        Return 0;
    }

    // 发单入口：开/平仓共用；无效价与挂单门禁在此拦截。
    // 分发必须用 If 块——IIF 两侧都会求值，会重复发单。
    Bool DoSendOrder(Integer orderDir, Integer lots, Numeric price, Bool isClose, String reason) {
        // 无效委托价兜底：NumValid 不通过（InvalidNumeric≈1.79e308 或 <=0）一律不发单——
        // 防历史回放/盘口未就绪时把无效价送进交易层（TBCore: trade Error Args）
        If (NumValid(price) == False) {
            If (LastBadPxYmd != YMD()) {
                LastBadPxYmd = YMD();
                WLogW("ORDER_SKIP 无效委托价 reason=" + reason
                    + " dir=" + Text(orderDir) + " lots=" + Text(lots) + " price=" + Text(price, 1));
            }
            Return False;
        }
        // 挂单门禁：存在未完成委托 → 一律不发新单（开/平/加/损同规则；回测集合恒空）
        If (GetArraySize(PendingIds) > 0) {
            If (LastGateYmd <> YMD()) {
                LastGateYmd = YMD();
                WLogW("ORDER_GATE 未完成委托=" + Text(GetArraySize(PendingIds))
                    + " ids=" + PendingStr() + "，禁发新单 reason=" + reason
                    + " dir=" + Text(orderDir) + " lots=" + Text(lots)
                    + " close=" + IIFString(isClose, "True", "False"));
            }
            Return False;
        }
        Bool ok = False;
        If (IsLiveTrade()) {
            // 模拟/实盘：账户接口（不进平台 bar 信号统计 → 不会信号闪烁/成交回滚）
            ok = SendAcctOrder(orderDir, lots, price, isClose, reason);
            Return ok;
        }
        If (isClose) {
            If (orderDir > 0) {
                ok = BuyToCover(lots, price);
            }
            Else {
                ok = Sell(lots, price);
            }
        }
        Else {
            If (orderDir > 0) {
                ok = Buy(lots, price);
            }
            Else {
                ok = SellShort(lots, price);
            }
        }
        LastSentLots = lots;
        If (ok == False) {
            WLogE(IIFString(isClose, "CLOSE_FAIL", "OPEN_FAIL") + " 委托被拒 reason=" + reason
                + " dir=" + Text(orderDir) + " lots=" + Text(lots) + " price=" + Text(price, 1));
        }
        Return ok;
    }

    // 开 1 个 unit 的公共实现：按预期成交价记账；开仓/加仓唯一出口。
    Integer DoOpenUnitCore(Integer dir, Numeric orderPrice, String reason) {
        // 授权有效期：只拦开仓/加仓，平仓与止损不受影响
        If (LicenseValid() == False) {
            If (LastLicenseWarnYmd <> IntPart(CurrentDate())) {
                LastLicenseWarnYmd = IntPart(CurrentDate());
                WLogE("LICENSE 已到期（有效至 " + Text(LicenseEndYmd())
                    + "），禁止开仓/加仓；止损与平仓保留");
            }
            Return 0;
        }
        If (ParamsOk == False) {
            Return 0;
        }
        Integer lots = CalcLots(Close());
        If (lots < 1) {
            // 节流：手数不足当日不会改善（逐 bar 重试均被拒），每天只记第一条防刷屏
            If (LastSkipYmd <> YMD()) {
                LastSkipYmd = YMD();
                WLogW("OPEN_SKIP 手数不足 reason=" + reason + " dir=" + Text(dir) + " price=" + Text(orderPrice, 1));
            }
            Return 0;
        }
        If (DoSendOrder(dir, lots, orderPrice, False, reason) == False) {
            Return 0;
        }
        Bool firstLeg = (PLots == 0);
        PLots = PLots + lots;
        PCost = PCost + orderPrice * lots;
        LastUnitPrice = orderPrice;
        UnitCount = UnitCount + 1;
        If (firstLeg) {
            TradeReason = reason;   // 首仓决定本笔入场类型；加仓不改变（供 S1 skip 武装判断）
        }
        PDir = dir;
        StopLine = LastUnitPrice - dir * StopDistN * NVal;
        WLog(reason + " OPEN dir=" + Text(dir) + " lots=" + Text(lots) + " price=" + Text(orderPrice, 1)
            + " N=" + Text(NVal, 1) + " stop=" + Text(StopLine, 1)
            + " units=" + Text(UnitCount) + " totalLots=" + Text(PLots) + " equity=" + Text(CurEquity(), 0));
        Return 0;
    }

    // 触发价入场（突破/加仓）：理论触发价 refPrice → CalcOrderPrice 解析实际可成交价
    Integer DoOpenUnit(Integer dir, Numeric refPrice, String reason) {
        Return DoOpenUnitCore(dir, CalcOrderPrice(dir, refPrice), reason);
    }

    // 市价入场（STALL 边沿）：回测=当前 1m 收盘价；实盘=盘口对手价（无效回退 Close）
    Integer DoOpenUnitMarket(Integer dir, String reason) {
        Numeric orderPrice = Close();
        If (IsLiveTrade()) {
            orderPrice = CalcOrderPrice(dir, Close());
        }
        Return DoOpenUnitCore(dir, orderPrice, reason);
    }

    // 台账/入场类型清零：OnInit 复位 / 平仓成功后 / 空仓残留清理共用
    Integer ResetLedger() {
        PLots = 0;
        PCost = 0;
        UnitCount = 0;
        LastUnitPrice = 0;
        StopLine = 0;
        PDir = 0;
        TradeReason = "";
        Return 0;
    }

    // 全部平仓；盈利时按 S1Scope × TradeReason 武装 skip，再清台账。
    Integer DoCloseAll(Integer dir, Numeric refPrice, String reason) {
        If (PLots <= 0) {
            Return 0;
        }
        // 委托方向 = 持仓方向取反（多平=卖、空平=买）
        Numeric orderPrice = CalcOrderPrice(dir * -1, refPrice);
        Numeric avg = PosAvgChart();
        If (avg <= 0 And PLots > 0) {
            avg = PCost / PLots;
        }
        Integer lots = PLots;
        If (DoSendOrder(dir * -1, lots, orderPrice, True, reason) == False) {
            Return 0;
        }
        // 账户路径会把平仓手数收敛为实际同向持仓（可能 ≠ 台账），盈亏按实发手数计
        Integer sent = LastSentLots;
        If (sent <= 0) {
            sent = lots;
        }
        Numeric pnl = (orderPrice - avg) * sent * MultiplierP * dir;
        WLog(reason + " CLOSE dir=" + Text(dir) + " lots=" + Text(sent) + " price=" + Text(orderPrice, 1)
            + " avg=" + Text(avg, 1) + " pnl=" + Text(pnl, 0)
            + " units=" + Text(UnitCount) + " equity=" + Text(CurEquity(), 0));
        If (S1Skip > 0 And pnl > 0) {
            // 武装范围对齐聚宽 s1_scope（默认 breakout）：按本笔首仓入场类型武装对应 skip；
            // STALL 在 breakout/breakout_nr 下不武装（current 下与其它入场同样武装）；武装后不分方向
            Bool armBreak = False;
            If (S1Scope == "current"
                Or ((S1Scope == "breakout" Or S1Scope == "breakout_nr") And TradeReason == "entry")) {
                armBreak = True;
            }
            Bool armNr = False;
            If (S1Scope == "current" Or (S1Scope == "breakout_nr" And TradeReason == "entry_nr")) {
                armNr = True;
            }
            If (armBreak) {
                SkipNextBreakout = True;
                SkipLevel = 0;
                SkipDir = 0;
            }
            If (armNr) {
                NrSkipNext = True;
            }
            String sideTxt = "long";
            If (dir < 0) {
                sideTxt = "short";
            }
            If (armBreak Or armNr) {
                String armedTxt = "";
                If (armBreak) {
                    armedTxt = "breakout";
                }
                If (armNr) {
                    If (armedTxt == "") {
                        armedTxt = "nr";
                    }
                    Else {
                        armedTxt = armedTxt + "+nr";
                    }
                }
                WLog("S1_SKIP_ARM side=" + sideTxt + " reason=" + TradeReason + " scope=" + S1Scope
                    + " armed=" + armedTxt + "（下一次同类入场跳过，不分方向）");
            }
            Else {
                WLog("S1_SKIP_NOARM side=" + sideTxt + " reason=" + TradeReason + " scope=" + S1Scope
                    + "（该入场类型不产生 skip）");
            }
        }
        ResetLedger();
        Return 0;
    }

    // ---------- 台账一致性：重启/部分成交后按平台持仓重建 ----------

    // 台账重建：以平台持仓为准整体覆盖（数量失配/部分成交/换月）。
    // StopLine 用重建时的 NVal 重推，ATR 已变时与原始止损有偏差。
    Integer RebuildLedger() {
        // 模拟/实盘：图表口径看不到映射仓，只能按账户真值重建
        If (IsLiveTrade()) {
            Return RebuildLedgerFromAcct();
        }
        // 平台真值：当前总手数（Abs 统一多空符号、IntPart 取整）
        Integer cc = IntPart(Abs(CurrentContracts()));
        PLots = cc;
        // 开仓次数 = unit 数（每次加仓是独立 entry）；Max 兜底：有持仓至少 1
        UnitCount = Max(1, CurrentEntries());
        // 平台成交加权均价；未就绪/异常（≤0）退回收盘价，避免 PCost 归零
        Numeric avg = AvgEntryPrice();
        If (avg <= 0) {
            avg = Close();
        }
        PCost = avg * cc;
        // 平台末次开仓成交价（加仓步进与共享止损的基准）；异常（≤0）退回均价
        LastUnitPrice = LastEntryPrice();
        If (LastUnitPrice <= 0) {
            LastUnitPrice = avg;
        }
        // 共享止损再推导：末 unit ± StopDistN×N（随持仓方向；用重建时的 N，见头注局限）
        Integer dir = 1;
        If (MarketPosition() < 0) {
            dir = -1;
        }
        PDir = dir;
        StopLine = LastUnitPrice - dir * StopDistN * NVal;
        // WARNING 留痕：contracts/entries/avg/lastEntry/stop 全量落日志，便于事后审计
        WLogW("STATE_REBUILD contracts=" + Text(cc) + " entries=" + Text(UnitCount) + " avg=" + Text(avg, 1)
            + " lastEntry=" + Text(LastUnitPrice, 1) + " stop=" + Text(StopLine, 1));
        Return 0;
    }

    // ---------- 主力移仓跟踪（原生自动换仓后重建基准）----------

    // 主力换月检测：MainSymbol 变化即换月；有持仓时重建台账。
    // 回测立即重建；模拟/实盘延后 2 根新 bar，由账户真值重建。
    Integer UpdateMainSymbol() {
        // 平台当前主力合约代码（主连/映射模式下随换月切换）
        String ms = MainSymbol();
        // 初始化分支：首个 bar 只记基线——重启后 LastMainSym 可能残留上一轮值，
        // MainSymInit=False 跳过比较，防启动瞬间误报
        If (MainSymInit == False) {
            MainSymInit = True;
            LastMainSym = ms;
            Return 0;
        }
        // 字符串变化 = 发生换月（自动换仓已在平台侧完成）
        If (ms <> LastMainSym) {
            // WARNING 留痕：旧→新 + 换仓后的台账/账户状态（pos=方向，lots=台账手数）
            WLogW("ROLLOVER " + LastMainSym + " -> " + ms + " pos=" + Text(PosDirNow())
                + " lots=" + Text(PosLotsNow()) + " avg=" + Text(PosAvgChart(), 1));
            LastMainSym = ms;
            If (PosLotsNow() <> 0) {
                If (IsLiveTrade()) {
                    // 换仓/委托映射切换需要时间：延后 2 根新 bar 再按账户真值对账，防读到换仓中间态
                    SyncBlockBars = Max(SyncBlockBars, 2);
                    WLogW("ROLLOVER_SYNC 延后 2 根新 bar 按账户真值重建");
                }
                Else {
                    // 有持仓：台账基准整体换成新合约口径（均价/末次入场价/止损线）
                    RebuildLedger();
                }
            }
        }
        Return 0;
    }

    // ---------- 信号逻辑 ----------

    // OnInit 时 data1 可能尚未就绪 → MappingOk 误锁。首个 data1 齐备的 bar 复核一次：
    // 数据层就绪后仍不符（合约错位/非主连/MapToMain=False）则保持锁定并记 ERROR（静态配置，不会运行中变化）。
    Integer RecheckMapping() {
        If (MappingRecheck == False) {
            Return 0;
        }
        If (StrategyMode() <> Enum_Mode_Strategy_Trade()) {
            MappingRecheck = False;
            Return 0;
        }
        If (DataCount() < 2) {
            Return 0;   // data1 尚未挂载/就绪，下一 bar 再试
        }
        MappingRecheck = False;
        Bool isMainCont = (Symbol == MainContSymbol);
        Bool layerOk = (D1Sym() == Symbol);
        If (isMainCont And layerOk And MapToMain) {
            MappingOk = True;
            WLog("MAPPING_RECHECK ok=True data0=" + Symbol + " data1=" + D1Sym()
                + "（OnInit 时 data1 未就绪，已解锁开仓）");
        }
        Else {
            WLogE("MAPPING_RECHECK ok=False data0=" + Symbol + " data1=" + D1Sym()
                + " mainC=" + MainContSymbol + " MapToMain=" + IIFString(MapToMain, "True", "False"));
        }
        Return 0;
    }

    // 交易统一门槛：挂图合法 + 模拟/实盘账户就绪 + 已到 TradeStartDate + N/四通道有效。
    Bool CanTradeNow() {
        // 模拟/实盘挂图不合法（非主连/两图层不一致/MapToMain=False）→ 禁止开新仓
        If (MappingOk == False) {
            Return False;
        }
        // 交易模式但账户数据未就绪 → 拒发一切委托（回测无此限制）
        If (IsStrategyTradeMode()) {
            If (IsLiveTrade() == False) {
                Return False;
            }
            // 账户持仓读取失败 → 无法做发送前门禁，禁开仓（止损平仓走 ManageStopOnly 兜底）
            If (AcctPosOk == False) {
                Return False;
            }
        }
        If (YMD() < TradeStartDate) {
            Return False;
        }
        If (NVal <= 0 Or EntryLongLv <= 0 Or EntryShortLv <= 0 Or ExitLongLv <= 0 Or ExitShortLv <= 0) {
            Return False;
        }
        Return True;
    }

    // 突破入场：空仓时检查 1m 价格是否触及 EntryN 通道上/下沿。
    // 系统1 跳过：被跳过价位同价不再入场，直到通道刷新出新极值。
    // 每 bar 至多开 1 unit；多头分支优先。
    Integer TryEntry() {
        // 参数校验失败：不开新仓，且不消费 SkipNextBreakout / 不写 BREAKOUT_SKIP
        If (ParamsOk == False) {
            Return 0;
        }
        // FixedDir 方向开关（对齐聚宽 fixed_dir）：0=双向 / 1=仅多 / -1=仅空
        Bool allowLong = (FixedDir == 0 Or FixedDir == 1);
        Bool allowShort = (FixedDir == 0 Or FixedDir == -1);
        // 多头突破
        If (allowLong And High() >= EntryLongLv) {
            // 系统1 跳过：记录被跳过价位，同价位不再入场（同聚宽 skip_level）
            If (S1Skip > 0 And SkipNextBreakout) {
                SkipNextBreakout = False;
                SkipLevel = EntryLongLv;
                SkipDir = 1;
                WLog("BREAKOUT_SKIP long level=" + Text(EntryLongLv, 1));
                Return 0;
            }
            If (S1Skip > 0 And SkipDir == 1 And SkipLevel > 0) {
                If (EntryLongLv == SkipLevel) {
                    Return 0;
                }
                WLog("SKIP_RELEASED long old=" + Text(SkipLevel, 1) + " new=" + Text(EntryLongLv, 1));
                SkipLevel = 0;
                SkipDir = 0;
            }
            // 触发价=通道上沿；实际委托价/手数在 DoOpenUnit 内解析
            DoOpenUnit(1, EntryLongLv, "entry");
            Return 0;
        }
        // 空头突破（多头已触发则上面已 Return——每 bar 单动作）
        If (allowShort And Low() <= EntryShortLv) {
            // 系统1 跳过：记录被跳过价位，同价位不再入场（同聚宽 skip_level）
            If (S1Skip > 0 And SkipNextBreakout) {
                SkipNextBreakout = False;
                SkipLevel = EntryShortLv;
                SkipDir = -1;
                WLog("BREAKOUT_SKIP short level=" + Text(EntryShortLv, 1));
                Return 0;
            }
            If (S1Skip > 0 And SkipDir == -1 And SkipLevel > 0) {
                If (EntryShortLv == SkipLevel) {
                    Return 0;
                }
                WLog("SKIP_RELEASED short old=" + Text(SkipLevel, 1) + " new=" + Text(EntryShortLv, 1));
                SkipLevel = 0;
                SkipDir = 0;
            }
            // 触发价=通道下沿；实际委托价/手数在 DoOpenUnit 内解析
            DoOpenUnit(-1, EntryShortLv, "entry");
        }
        Return 0;
    }

    // ---------- STALL 极值停滞并行入场（v3.2；口径=data1 周期，默认日线）----------

    // STALL 运行状态复位（OnInit 调用；StallEnabled 随后由 StallPeriod 校验重新置位）
    Integer ResetStallState() {
        StallEnabled = False;
        StallInit = False;
        StallSigO = 0;
        StallSigH = 0;
        StallSigL = 0;
        StallSigC = 0;
        StallLongCnt = -1;
        StallShortCnt = -1;
        StallRefSym = "";
        Return 0;
    }

    // 计数：最新完成 data1 bar 距最近一次“创新低/新高”事件的 bar 数（事件当根=0；无事件=-1）。
    // 事件 := Low(j) < min(Low(j+1..j+n))（多头）/ High(j) > max(High(j+1..j+n))（空头），严格不等。
    // 与聚宽 stall_state 等价：j 对应 i=last-j+1，计数 last-i=j-1；扫描窗口 2n+5 根。
    Integer CalcStallCounts(Integer n) {
        StallLongCnt = -1;
        StallShortCnt = -1;
        Integer lim = n + 5;
        Integer j;
        Integer k;
        Numeric ext;
        For j = 1 To lim {
            If (StallLongCnt < 0) {
                ext = D1Low(j + 1);
                For k = 1 To n {
                    ext = Min(ext, D1Low(j + k));
                }
                If (D1Low(j) < ext) {
                    StallLongCnt = j - 1;
                }
            }
            If (StallShortCnt < 0) {
                ext = D1High(j + 1);
                For k = 1 To n {
                    ext = Max(ext, D1High(j + k));
                }
                If (D1High(j) > ext) {
                    StallShortCnt = j - 1;
                }
            }
        }
        Return 0;
    }

    // STALL 边沿评判与入场。返回 True = 本 bar STALL 分支已占用。
    // 边沿 = Data1[1] OHLC 签名变化；首次仅建基线，持仓/挂单/预热期达成不补。
    Bool TryStallEntry(Bool allowOrder) {
        // 参数校验失败：不开新仓（基线推进一并无意义——参数整轮不变）
        If (ParamsOk == False) {
            Return False;
        }
        // StallEnabled 已由 OnInit 统一校验（StallPeriod<=0 或窗口超限 → False）
        If (StallEnabled == False) {
            Return False;
        }
        If (DataCount() < 2) {
            Return False;
        }
        // Data1[1] 四价有效性逐项校验（NumValid：无效值或 <=0 均拒绝；Or 不依赖短路）
        If (NumValid(D1Open(1)) == False Or NumValid(D1High(1)) == False Or
            NumValid(D1Low(1)) == False Or NumValid(D1Close(1)) == False) {
            Return False;
        }
        Numeric so = D1Open(1);
        Numeric sh = D1High(1);
        Numeric sl = D1Low(1);
        Numeric sc = D1Close(1);
        String ref = D1Sym() + "|" + MainSymbol();
        // 首根 / 换月换层：仅建基线，不触发
        If (StallInit == False Or StallRefSym <> ref) {
            StallRefSym = ref;
            StallSigO = so;
            StallSigH = sh;
            StallSigL = sl;
            StallSigC = sc;
            StallInit = True;
            Return False;
        }
        // 同一根 data1 bar 已评判过
        If (so == StallSigO And sh == StallSigH And sl == StallSigL And sc == StallSigC) {
            Return False;
        }
        // 新 data1 bar：先消费边沿，再同步 N/四通道（OnBarOpen 可能仍读到旧 Data1[1]）
        StallSigO = so;
        StallSigH = sh;
        StallSigL = sl;
        StallSigC = sc;
        RefreshNChannels();
        // 回放期：边沿已消费、通道已同步，但不下单（OnBar 已跳过 ManagePosition）
        If (allowOrder == False) {
            Return False;
        }
        If (CanTradeNow() == False) {
            If (LogLevel >= 3) {
                WLogD("STALL_MISS reason=gated");
            }
            Return False;
        }
        CalcStallCounts(StallPeriod);
        If (LogLevel >= 3) {
            WLogD("STALL_EDGE sig=" + Text(so, 1) + "/" + Text(sh, 1) + "/" + Text(sl, 1) + "/" + Text(sc, 1)
                + " long=" + Text(StallLongCnt) + " short=" + Text(StallShortCnt));
        }
        // FixedDir=0 且多空同时达成：放弃本次（同聚宽 stall_edge；避免固定偏向多头）
        If (FixedDir == 0 And StallLongCnt == StallPeriod And StallShortCnt == StallPeriod) {
            If (LogLevel >= 3) {
                WLogD("STALL_MISS reason=tie long=" + Text(StallLongCnt) + " short=" + Text(StallShortCnt));
            }
            Return False;
        }
        Bool allowLong = (FixedDir == 0 Or FixedDir == 1);
        Bool allowShort = (FixedDir == 0 Or FixedDir == -1);
        Integer side = 0;
        If (allowLong And StallLongCnt == StallPeriod) {
            side = 1;
        }
        Else {
            If (allowShort And StallShortCnt == StallPeriod) {
                side = -1;
            }
        }
        If (side == 0) {
            Return False;
        }
        String sideTxt = "long";
        If (side < 0) {
            sideTxt = "short";
        }
        If (PosLotsNow() <> 0) {
            WLog("STALL_MISS side=" + sideTxt + " reason=holding stall=" + Text(StallPeriod));
            Return False;
        }
        // 已有未完成委托（开/平/加/损）→ 本边沿放弃，不补触发；本 bar 也不再跑常规检查
        // （同聚宽：stall 分支命中后 open_unit 被挂单拦截即 return，不落回突破/加仓判定）
        If (GetArraySize(PendingIds) > 0) {
            WLog("STALL_MISS side=" + sideTxt + " reason=pending ids=" + PendingStr());
            Return True;
        }
        // 平台空仓但台账残留：先清再开（与 ManagePosition 空仓分支同口径）
        If (PLots <> 0 Or UnitCount <> 0) {
            WLogW("LEDGER_CLEAN 空仓残留台账清理 prevLots=" + Text(PLots));
            ResetLedger();
        }
        WLog("STALL_FIRE side=" + sideTxt + " stall=" + Text(StallPeriod)
            + " long=" + Text(StallLongCnt) + " short=" + Text(StallShortCnt)
            + " N=" + Text(NVal, 1));
        // 无论手数不足/委托被拒，均视为本 bar 已尝试（对齐聚宽：STALL 分支不再跑海龟检查）
        DoOpenUnitMarket(side, "entry_stall");
        Return True;
    }

    // ---------- NR 收缩并行入场（v3.2；口径=data1 周期）----------

    // NR 运行状态复位（OnInit 调用；NrEnabled 随后由 NrN 校验重新置位）
    Integer ResetNrState() {
        NrEnabled = False;
        NrHasEvent = False;
        NrFired = False;
        NrSkipNext = False;
        NrHighLv = 0;
        NrLowLv = 0;
        NrD1Sig = "";
        Return 0;
    }

    // 刷新 NR 事件：最新完成 data1 bar 的 OHLC 指纹变化时重算。
    // 事件 := 该 bar 振幅（High-Low）为近 NrN 根最小（含等号，同聚宽）；触发沿 = 该 bar High/Low。
    // 事件在下一根 data1 bar 期间有效；NrFired 每根新 data1 bar 复位。
    Integer RefreshNrEvent() {
        If (NrEnabled == False Or DataCount() < 2) {
            NrHasEvent = False;
            Return 0;
        }
        If (NumValid(D1Open(1)) == False Or NumValid(D1High(1)) == False Or
            NumValid(D1Low(1)) == False Or NumValid(D1Close(1)) == False) {
            NrHasEvent = False;
            Return 0;
        }
        Numeric o = D1Open(1);
        Numeric h = D1High(1);
        Numeric l = D1Low(1);
        Numeric c = D1Close(1);
        String sig = D1Sym() + "|" + MainSymbol() + "|" + Text(o, 4) + "|" + Text(h, 4)
            + "|" + Text(l, 4) + "|" + Text(c, 4);
        If (sig == NrD1Sig) {
            Return 0;
        }
        NrD1Sig = sig;
        Numeric lastR = h - l;
        Numeric minR = lastR;
        Numeric r;
        Integer k;
        Integer validCnt = 1;
        For k = 2 To NrN {
            If (NumValid(D1High(k)) And NumValid(D1Low(k))) {
                r = D1High(k) - D1Low(k);
                minR = Min(minR, r);
                validCnt = validCnt + 1;
            }
        }
        NrHighLv = h;
        NrLowLv = l;
        NrFired = False;
        // 窗口不足 NrN 根：不构成事件（等历史补足）
        If (validCnt < NrN) {
            NrHasEvent = False;
            Return 0;
        }
        NrHasEvent = (lastR <= minR);
        If (NrHasEvent And LogLevel >= 3) {
            WLogD("NR_SET n=" + Text(NrN) + " high=" + Text(h, 1) + " low=" + Text(l, 1));
        }
        Return 0;
    }

    // NR 突破入场。返回 True = 本 bar 已按 NR 处理（开仓/跳过/挂单暂缓）。
    // 开仓成功才置 NrFired；手数不足/拒单下一分钟重试。
    Bool TryNrEntry() {
        Integer prevLots = 0;
        If (ParamsOk == False Or NrEnabled == False Or NrHasEvent == False Or NrFired) {
            Return False;
        }
        If (PosLotsNow() <> 0) {
            Return False;
        }
        // 与 STALL 同口径：挂图/账户/交易起点/N 通道不满足时不动作，事件保留待后续 bar
        If (CanTradeNow() == False) {
            If (LogLevel >= 3) {
                WLogD("NR_MISS reason=gated");
            }
            Return False;
        }
        Bool allowLong = (FixedDir == 0 Or FixedDir == 1);
        Bool allowShort = (FixedDir == 0 Or FixedDir == -1);
        // 多头
        If (allowLong And High() >= NrHighLv) {
            If (S1Skip > 0 And NrSkipNext And (S1Scope == "current" Or S1Scope == "breakout_nr")) {
                NrSkipNext = False;
                NrFired = True;
                WLog("NR_SKIP side=long level=" + Text(NrHighLv, 1) + "（系统1：上次交易盈利）");
                Return True;
            }
            If (GetArraySize(PendingIds) > 0) {
                If (LogLevel >= 3) {
                    WLogD("NR_MISS side=long reason=pending ids=" + PendingStr());
                }
                // 本 bar 仍按 NR 处理（不评估突破，同聚宽 open_unit 被挂单拦截后直接 return）；
                // 事件不消费，挂单解除后仍可入场
                Return True;
            }
            If (PLots <> 0 Or UnitCount <> 0) {
                WLogW("LEDGER_CLEAN 空仓残留台账清理 prevLots=" + Text(PLots));
                ResetLedger();
            }
            prevLots = PLots;
            DoOpenUnitMarket(1, "entry_nr");
            If (PLots > prevLots) {
                NrFired = True;
                WLog("NR_FIRE side=long n=" + Text(NrN) + " level=" + Text(NrHighLv, 1)
                    + " N=" + Text(NVal, 1));
            }
            Return True;
        }
        // 空头
        If (allowShort And Low() <= NrLowLv) {
            If (S1Skip > 0 And NrSkipNext And (S1Scope == "current" Or S1Scope == "breakout_nr")) {
                NrSkipNext = False;
                NrFired = True;
                WLog("NR_SKIP side=short level=" + Text(NrLowLv, 1) + "（系统1：上次交易盈利）");
                Return True;
            }
            If (GetArraySize(PendingIds) > 0) {
                If (LogLevel >= 3) {
                    WLogD("NR_MISS side=short reason=pending ids=" + PendingStr());
                }
                // 本 bar 仍按 NR 处理（不评估突破，同聚宽 open_unit 被挂单拦截后直接 return）；
                // 事件不消费，挂单解除后仍可入场
                Return True;
            }
            If (PLots <> 0 Or UnitCount <> 0) {
                WLogW("LEDGER_CLEAN 空仓残留台账清理 prevLots=" + Text(PLots));
                ResetLedger();
            }
            prevLots = PLots;
            DoOpenUnitMarket(-1, "entry_nr");
            If (PLots > prevLots) {
                NrFired = True;
                WLog("NR_FIRE side=short n=" + Text(NrN) + " level=" + Text(NrLowLv, 1)
                    + " N=" + Text(NVal, 1));
            }
            Return True;
        }
        Return False;
    }

    // data0 心跳（仅实时 OnBarClose，每根一行）：证明事件循环在跑，并留下价格/仓位/N 状态。
    // 有 stop 才附 stop，有挂单才附 pend，保持短行。
    Integer LogBarHeartbeat() {
        String stopSeg = "";
        If (PosLotsNow() <> 0) {
            stopSeg = " stop=" + Text(StopLine, 1);
        }
        String pendSeg = "";
        If (GetArraySize(PendingIds) > 0) {
            pendSeg = " pend=" + Text(GetArraySize(PendingIds));
        }
        // 模拟/实盘附加账户真值（图表口径看不到映射仓，pos= 恒 0 属正常）
        String acctSeg = "";
        If (IsLiveTrade()) {
            RefreshAccountPosition();
            acctSeg = " pdir=" + Text(PDir) + " acctL=" + Text(IntPart(AcctLong))
                + " acctS=" + Text(IntPart(AcctShort))
                + " acctOk=" + IIFString(AcctPosOk, "True", "False");
        }
        WLog("HB " + Symbol + " C=" + Text(Close(), 1)
            + " N=" + Text(NVal, 2)
            + " pos=" + Text(MarketPosition())
            + " units=" + Text(UnitCount) + " lots=" + Text(PLots) + stopSeg + pendSeg + acctSeg);
        Return 0;
    }

    // data1 完整行（仅实时 OnBarClose，OnBar 之后调用）：新完成的一根 data1 bar 一行，
    // 含 data1 OHLC + N/四通道 + STALL 计数 + 持仓台账。
    // 用单一字符串指纹去重（合约|OHLC）：同一根 data1 bar 只打印一次。
    Integer LogData1Bar() {
        If (DataCount() < 2) {
            Return 0;
        }
        // 四价有效性逐项校验（NumValid：无效值或 <=0 均拒绝；Or 不依赖短路）
        If (NumValid(D1Open(1)) == False Or NumValid(D1High(1)) == False Or
            NumValid(D1Low(1)) == False Or NumValid(D1Close(1)) == False) {
            Return 0;
        }
        Numeric o = D1Open(1);
        Numeric h = D1High(1);
        Numeric l = D1Low(1);
        Numeric c = D1Close(1);
        String sig = D1Sym() + "|" + MainSymbol() + "|" + Text(o, 4) + "|" + Text(h, 4)
            + "|" + Text(l, 4) + "|" + Text(c, 4);
        If (sig == D1LogSig) {
            Return 0;
        }
        D1LogSig = sig;
        WLog("D1BAR " + D1Sym() + "(" + D1Freq() + ") O=" + Text(o, 1)
            + " H=" + Text(h, 1) + " L=" + Text(l, 1) + " C=" + Text(c, 1)
            + " N=" + Text(NVal, 2)
            + " EL=" + Text(EntryLongLv, 1) + " XL=" + Text(ExitLongLv, 1)
            + " ES=" + Text(EntryShortLv, 1) + " XS=" + Text(ExitShortLv, 1)
            + " stallL=" + Text(StallLongCnt) + " stallS=" + Text(StallShortCnt)
            + " units=" + Text(UnitCount) + " lots=" + Text(PLots) + " stop=" + Text(StopLine, 1));
        Return 0;
    }

    // 交易主流程：每 bar 至多一个动作。空仓→突破入场；
    // 持仓→止损 > 反向突破出场 > 加仓，任一动作后立即返回。
    Integer ManagePosition() {
        // 空仓分支：先清残留台账（手工平仓/异常后可能残留），再尝试入场
        If (PosLotsNow() == 0) {
            If (PLots <> 0 Or UnitCount <> 0) {
                WLogW("LEDGER_CLEAN 空仓残留台账清理 prevLots=" + Text(PLots));
                ResetLedger();
            }
            TryEntry();
            Return 0;
        }
        // 持仓对账：平台手数 ≠ 台账 → 以平台为准重建（重启/部分成交/换月后；模拟/实盘走账户真值）
        If (PosLotsNow() <> PLots) {
            RebuildLedger();
        }
        // 持仓方向：+1 多 / -1 空（模拟/实盘=台账方向；回测=图表方向）
        Integer dir = PosDirNow();
        If (dir == 0) {
            Return 0;
        }
        // 1) 共享止损：多单 Low ≤ StopLine、空单 High ≥ StopLine（2N 距离，加仓后随末 unit 上移）
        If (StopLine > 0) {
            If (dir > 0 And Low() <= StopLine) {
                DoCloseAll(dir, StopLine, "stop");
                Return 0;
            }
            If (dir < 0 And High() >= StopLine) {
                DoCloseAll(dir, StopLine, "stop");
                Return 0;
            }
        }
        // 2) 反向突破出场：多单跌破 ExitLongLv、空单涨破 ExitShortLv（ExitN 通道反向沿）
        If (dir > 0 And Low() <= ExitLongLv) {
            DoCloseAll(dir, ExitLongLv, "exit_breakout");
            Return 0;
        }
        If (dir < 0 And High() >= ExitShortLv) {
            DoCloseAll(dir, ExitShortLv, "exit_breakout");
            Return 0;
        }
        // 3) 加仓：顺向每 AddStepN*N 加 1 unit；UnitCount < MaxUnits 封顶
        If (UnitCount < MaxUnits) {
            If (dir > 0 And High() >= LastUnitPrice + AddStepN * NVal) {
                DoOpenUnit(dir, LastUnitPrice + AddStepN * NVal, "add");
                Return 0;
            }
            If (dir < 0 And Low() <= LastUnitPrice - AddStepN * NVal) {
                DoOpenUnit(dir, LastUnitPrice - AddStepN * NVal, "add");
                Return 0;
            }
        }
        Return 0;
    }

    // data1 失效时的仅止损兜底（模拟/实盘）：只用最后有效 StopLine 保护持仓。
    Integer ManageStopOnly() {
        If (PosLotsNow() == 0 Or PLots <= 0 Or StopLine <= 0) {
            Return 0;
        }
        Integer dir = PosDirNow();
        If (dir == 0) {
            Return 0;
        }
        If (dir > 0 And Low() <= StopLine) {
            WLogW("STOP_ONLY data1无效，按最后有效止损线 " + Text(StopLine, 1) + " 平多");
            DoCloseAll(dir, StopLine, "stop");
            Return 0;
        }
        If (dir < 0 And High() >= StopLine) {
            WLogW("STOP_ONLY data1无效，按最后有效止损线 " + Text(StopLine, 1) + " 平空");
            DoCloseAll(dir, StopLine, "stop");
            Return 0;
        }
        Return 0;
    }

    OnInit() {
        // 全局状态复位（所有模式日志只追加、不删除）
        BarCnt = 0;
        MainSymInit = False;
        SkipNextBreakout = False;
        SkipLevel = 0;
        SkipDir = 0;
        ReplayGateLogged = False;
        RealtimeLogged = False;
        ResetStallState();
        ResetNrState();
        ResetLogThrottle();
        D1LogSig = "";
        FinalDone = False;
        FillCnt = 0;
        OrderCnt = 0;
        PendingClear();                // 挂单门禁集合不能跨次运行残留（orderId 会复用）
        MappingOk = False;
        MappingRecheck = False;
        ParamsOk = False;
        CalcOk = False;
        ResetChannels();
        ResetLedger();
        // 账户真值对账状态复位（模拟/实盘；回测不用）——首根实时 bar 由 SyncAcctOnNewBar 建基线
        AcctPosOk = False;
        AcctLong = 0;
        AcctShort = 0;
        AcctLongAvg = 0;
        AcctShortAvg = 0;
        AcctSym = "";
        SyncBlockBars = 0;
        LastSyncBarSig = "";
        LastSentLots = 0;
        // SetInitCapital 无返回值，合法性由 InitCapital>0 校验
        SetInitCapital(InitCapital);
        Bool okMargin = SetMarginRate(MarginRateP);
        Bool okComm = SetCommissionRate(Enum_Rate_ByFillAmount(), CommissionRateP);
        Bool okSlip = SetSlippage(Enum_Rate_PointPerHand(), SlipTicks);
        SetConsecEntries(-1);          // 平台不限建仓次数，加仓由 UnitCount<MaxUnits 控制
        ParamsOk = ValidateParams();
        If (ParamsOk) {
            WLogInit("参数校验: OK");
        }
        Else {
            WLogInitE("参数校验失败：禁止开新仓（含加仓）；止损/平仓保留");
        }
        // STALL 参数校验：2*StallPeriod+5 必须 <= WinSize()
        If (StallPeriod > 0) {
            If (StallPeriod * 2 + 5 > WinSize()) {
                WLogInitE("STALL 停用: 2*StallPeriod+5=" + Text(StallPeriod * 2 + 5)
                    + " > WinSize()=" + Text(WinSize()) + "（默认 NPeriod=20 时 StallPeriod 最大 28）");
            }
            Else {
                StallEnabled = True;
            }
        }
        // NR 参数校验：NrN<2 或 NrN>WinSize 时停用
        If (NrN > 0) {
            If (NrN < 2) {
                WLogInitE("NR 停用: NrN=" + Text(NrN) + "（需 >=2；与聚宽 win>=2 守卫一致）");
            }
            Else {
                If (NrN > WinSize()) {
                    WLogInitE("NR 停用: NrN=" + Text(NrN) + " > WinSize()=" + Text(WinSize()));
                }
                Else {
                    NrEnabled = True;
                }
            }
        }
        // 模拟/实盘：data0/data1 必须同为 MainContSymbol 且 MapToMain=True；回测不校验
        Bool isMainCont = False;
        Bool layerOk = False;
        MappingOk = True;
        If (StrategyMode() == Enum_Mode_Strategy_Trade()) {
            MappingOk = False;
            isMainCont = (Symbol == MainContSymbol);
            If (DataCount() >= 2) {
                If (D1Sym() == Symbol) {
                    layerOk = True;
                }
            }
            If (isMainCont And layerOk And MapToMain) {
                MappingOk = True;
            }
            // data1 可能晚于 OnInit 就绪 → 首个数据齐备的 bar 复核一次（仅解锁，不重设映射标志）
            If (MappingOk == False) {
                MappingRecheck = True;
            }
        }
        // 换月全套只在主连图/模拟/实盘（MapToMain=True）开启；单合约切片保持不复权
        Range[0:DataCount-1] {
            If (MapToMain And (StrategyMode() <> Enum_Mode_Strategy_Trade() Or isMainCont)) {
                AddDataFlag(Enum_Data_RolloverBackWard());
                AddDataFlag(Enum_Data_RolloverRealPrice());
                AddDataFlag(Enum_Data_AutoSwapPosition());
                SetSwapPosVolType(2);
                SetOrderMap2MainSymbol();
            }
        }
        WLogInit("===== OnInit 棕榈油海龟 v3.2（TB公式版-双图层+STALL+NR）=====");
        If (StrategyMode() <> Enum_Mode_Strategy_Trade()) {
            WLogInit("LICENSE 非模拟/实盘模式不校验（模拟/实盘有效至 " + Text(LicenseEndYmd()) + "）");
        }
        Else If (LicenseValid()) {
            WLogInit("LICENSE 有效至 " + Text(LicenseEndYmd()) + "（到期后禁止开仓/加仓；止损与平仓保留）");
        }
        Else {
            WLogInitE("LICENSE 已到期（有效至 " + Text(LicenseEndYmd()) + "）：禁止开仓/加仓；止损与平仓保留");
        }
        WLogInit("图层: DataCount=" + Text(DataCount()) + " MainSymbol=" + MainSymbol());
        WLogInit("data0.Symbol=" + Symbol + "(" + Frequency() + ")");
        If (DataCount() >= 2) {
            WLogInit("data1.Symbol=" + D1Sym() + "(" + D1Freq() + ")" + "（须与 data0 同合约；标准=日线，或其它周期）");
            // 仅首屏告警；运行期同合约守卫由 RefreshNChannels 执行
            If (D1Sym() <> Symbol) {
                WLogInitE("数据源错位: data1.Symbol=" + D1Sym() + " ≠ data0.Symbol=" + Symbol + "（图层顺序/合约不符，检查图表数据源）");
            }
        }
        Else {
            WLogInitE("只有 1 个数据源——v3.2 依赖 data1 图层：请先在图表右键追加同合约 data1（标准 1d）再运行");
        }
        WLogInit("合约: Symbol=" + Symbol + " Name=" + SymbolName() + " ContractUnit=" + Text(ContractUnit())
            + " BigPointValue=" + Text(BigPointValue(), 1)
            + " MinMove=" + Text(MinMove()) + " PriceScale=" + Text(PriceScale(), 2)
            + " MarginRatio=" + Text(MarginRatio(), 3));
        If (StrategyMode() == Enum_Mode_Strategy_Trade()) {
            WLogInit("MAPPING: data0=" + Symbol + " layers=" + Text(DataCount())
                + " data1Same=" + IIFString(layerOk, "True", "False")
                + " isMainCont=" + IIFString(isMainCont, "True", "False")
                + " MapToMain=" + IIFString(MapToMain, "True", "False")
                + " mainC=" + MainContSymbol
                + " ok=" + IIFString(MappingOk, "True", "False"));
            If (MappingOk == False) {
                If (isMainCont == False) {
                    WLogInitE("MAP_FAIL data0.Symbol=" + Symbol + " 不是主连合约（期望 " + MainContSymbol + "）；模拟/实盘要求 data0/data1 均为同一主连，MapToMain=True");
                }
                If (layerOk == False) {
                    If (DataCount() >= 2) {
                        WLogInitE("MAP_FAIL data1 与 data0 不一致（data0=" + Symbol + "，data1=" + D1Sym() + "）；模拟/实盘要求两图层同一主连合约");
                    }
                    Else {
                        WLogInitE("MAP_FAIL data1 图层缺失；模拟/实盘要求 data0/data1 均为同一主连合约");
                    }
                }
                If (MapToMain == False) {
                    WLogInitE("MAP_FAIL 模拟/实盘 MapToMain 必须 True（当前 False）→ 禁止开新仓");
                }
            }
        }
        Else {
            WLogInit("MAPPING: 回测模式不校验（主连回测建议 MapToMain=True）");
        }
        WLogInit("模式: StrategyMode=" + Text(StrategyMode())
            + " IsLiveTrade=" + IIFString(IsLiveTrade(), "True", "False")
            + " AutoTrade=" + IIFString(IsTradeEnabled(), "True", "False")
            + " Trade=" + Text(Enum_Mode_Strategy_Trade()) + " Chart=" + Text(Enum_Mode_Strategy_Chart()));
        WLogInit("GATE: 挂单门禁=OnOrder 未完成委托集合 + 每根 bar 按平台未成交列表校正；回放期门禁=模拟/实盘非实时 bar 不下单；图表模式自动交易开（图表交易）一律不发单；简单回测与图表回测不拦");
        WLogInit("ORDER_CH: 模拟/实盘=A_*账户接口（落地合约+真实价/图价÷Rollover，无信号统计→不闪烁；每根实时新 bar 4 参 A_GetPosition 对账）"
            + " 回测=Buy/SellShort/Sell/BuyToCover（图表信号）");
        WLogInit("参数A: FixedDir=" + Text(FixedDir) + " NPeriod=" + Text(NPeriod) + " NSmooth=" + Text(NSmooth)
            + " EntryN=" + Text(EntryN) + " ExitN=" + Text(ExitN) + " StallPeriod=" + Text(StallPeriod)
            + " NrN=" + Text(NrN) + " MaxUnits=" + Text(MaxUnits));
        String stallFreq = "na";
        If (DataCount() >= 2) {
            stallFreq = D1Freq();
        }
        WLogInit("STALL: enabled=" + IIFString(StallEnabled, "True", "False")
            + " StallPeriod=" + Text(StallPeriod) + " win=" + Text(WinSize())
            + " need=" + Text(StallPeriod * 2 + 5) + " data1周期=" + stallFreq);
        WLogInit("NR: enabled=" + IIFString(NrEnabled, "True", "False")
            + " NrN=" + Text(NrN) + " win=" + Text(WinSize()) + " data1周期=" + stallFreq);
        WLogInit("参数B: UnitRisk=" + Text(UnitRisk, 4) + " MarginCapPct=" + Text(MarginCapPct, 4)
            + " AddStepN=" + Text(AddStepN, 2) + " StopDistN=" + Text(StopDistN, 2)
            + " S1Skip=" + Text(S1Skip) + " S1Scope=" + S1Scope);
        WLogInit("参数C: TradeStartDate=" + Text(TradeStartDate) + " InitCapital=" + Text(InitCapital, 0)
            + " MultiplierP=" + Text(MultiplierP, 2) + " MarginRateP=" + Text(MarginRateP, 4)
            + " CommissionRateP=" + Text(CommissionRateP, 8) + " SlipTicks=" + Text(SlipTicks, 2)
            + " RealTickSize=" + Text(RealTickSize, 2));
        WLogInit("日志: LogDir=[" + LogDir + "] 实际目录=" + IIFString(LogDir == "", "(空=平台运行目录)", EffectiveLogDir())
            + " 文件=" + LogPath());
        WLogInit("设置(原始返回): SetMarginRate=" + IIFString(okMargin, "True", "False")
            + " SetCommissionRate=" + IIFString(okComm, "True", "False")
            + " SetSlippage=" + IIFString(okSlip, "True", "False")
            + " MapToMain=" + IIFString(MapToMain, "True(OrderMap2Main,RolloverOn)", "False(OrderMapOff,RolloverOff)"));
    }

    // 新 bar 生成时驱动（历史回放与实盘均驱动）
    OnBarOpen(ArrayRef<Integer> indexs) {
        // [红线] 只放 Range[0:0]，避免多图层各跑一遍
        Range[0:0] {
            // N/四通道每根新 bar 刷新一次；失败归 0 拒交易，下根自动重试
            RefreshNChannels();
            // 模拟/实盘实时：按账户真值对账（成交/换月后宽限 2 根）
            If (IsLiveTrade() And QuoteStatus() == Enum_QuoteStatus_RealTime) {
                SyncAcctOnNewBar();
            }
        }
    }

    OnBar(ArrayRef<Integer> indexs) {
        // [红线] 只放 Range[0:0]：每个图层各跑一遍会导致双跑
        Range[0:0] {
            BarCnt = BarCnt + 1;

            // 首 bar 再清一次挂单集合，防上次运行 orderId 残留
            If (BarCnt == 1) {
                PendingClear();
            }
            // 每根 bar 按平台未成交列表校正挂单集合（模拟/实盘）
            SyncPendingFromAccount();

            // 首 bar 警告：实盘未到 TradeStartDate
            If (BarCnt == 1 And IsLiveTrade() And YMD() < TradeStartDate) {
                WLogW("TRADE_GATE 实盘未到 TradeStartDate=" + Text(TradeStartDate) + "，不会产生交易动作");
            }

            // ---- 1. 主力移仓检测（原生自动换仓）----
            UpdateMainSymbol();
            // 1b. 挂图合法性复核：OnInit 时 data1 可能未就绪，首个数据齐备的 bar 解锁一次
            RecheckMapping();

            // ---- 2. NR 事件刷新（只更新事件状态；入场在 STALL 之后判定）----
            RefreshNrEvent();

            // ---- 3. STALL / NR 并行入场（先于常规海龟检查）----
            // 非实时 bar（历史回放）只推进基线与通道，不下单；回测不受影响。
            If (IsTradablePhase() == False) {
                If (ReplayGateLogged == False) {
                    ReplayGateLogged = True;
                    WLog("REPLAY_GATE 跳过交易：只更新通道/STALL/NR 基线（mode=" + Text(StrategyMode())
                        + " QuoteStatus=" + Text(QuoteStatus()) + "）");
                }
                TryStallEntry(False);
            }
            Else {
                If (IsStrategyTradeMode()) {
                    If (RealtimeLogged == False) {
                        RealtimeLogged = True;
                        WLog("REALTIME_BAR QuoteStatus=" + Text(QuoteStatus()) + " 恢复交易");
                    }
                }
                // True=边沿命中，STALL 分支已占用；否则再判 NR，最后走常规检查
                If (TryStallEntry(True) == False) {
                    // NR 已处理（触发/跳过/挂单暂缓）→ 本 bar 不再走 ManagePosition
                    If (TryNrEntry() == False) {
                        // 风控与信号（N/四通道无效期 levels=0 → CanTradeNow=False，不下单）
                        If (CanTradeNow()) {
                            ManagePosition();
                        }
                        Else {
                            // 交易模式但账户数据未就绪：暂停一切委托（CanTradeNow 已拦），每交易日首条告警
                            If (IsStrategyTradeMode()) {
                                If (IsLiveTrade() == False) {
                                    If (YMD() >= TradeStartDate) {
                                        If (LastAcctWarnYmd <> YMD()) {
                                            LastAcctWarnYmd = YMD();
                                            WLogW("ACCT_NOT_READY 交易模式但账户数据未就绪，暂停交易（检查账户连接/绑定）");
                                        }
                                    }
                                }
                            }
                            // data1 失效期：实盘/模拟仍用最后有效止损线保护持仓；
                            // 回测保持原语义，TradeStartDate 之前也不动作。
                            If (IsLiveTrade() And YMD() >= TradeStartDate) {
                                ManageStopOnly();
                            }
                        }
                    }
                }
            }

            // ---- 4. 每日定时日志（14:45）----
            // 回测照写；模拟/实盘仅实时
            Bool dailyLogOk = True;
            If (IsStrategyTradeMode()) {
                dailyLogOk = (QuoteStatus() == Enum_QuoteStatus_RealTime);
            }
            If (dailyLogOk And Hour() == LogHour And Minute() == LogMinute And YMD() <> LastLogYmd) {
                LastLogYmd = YMD();
                String d1s = "na";
                String d0seg = "";
                String pendseg = "";
                If (DataCount() >= 2) {
                    d1s = Text(D1Close(1), 1);
                    // d0diag 仅 DEBUG 级：读 Data1[0] 禁区仅供诊断，禁入判断
                    If (LogLevel >= 3) {
                        d0seg = " d0diag=" + Text(D1Close0Diag(), 1);
                    }
                }
                // 挂单门禁状态：仅在存在未完成委托时附加（回测集合恒空）
                If (GetArraySize(PendingIds) > 0) {
                    pendseg = " pend=" + Text(GetArraySize(PendingIds)) + " ids=" + PendingStr();
                }
                // 合约标签：前缀=指标来源 data1（无 data1 时退回 data0）
                String symseg = Symbol;
                If (DataCount() >= 2) {
                    symseg = D1Sym();
                }
                // 账户真值（模拟/实盘）：每天一行，供事后核对台账 vs 账户；
                // 落地合约与 data1 不同（连续图→真实主力）时附 acctSym，避免把连续合约当成可交易合约
                String acctseg = "";
                If (IsLiveTrade()) {
                    RefreshAccountPosition();
                    If (AcctSym <> "" And AcctSym <> symseg) {
                        acctseg = acctseg + " acctSym=" + AcctSym;
                    }
                    acctseg = acctseg + " pdir=" + Text(PDir) + " acctL=" + Text(IntPart(AcctLong))
                        + " acctS=" + Text(IntPart(AcctShort))
                        + " acctAvgL=" + Text(AcctLongAvg, 1) + " acctAvgS=" + Text(AcctShortAvg, 1)
                        + " acctOk=" + IIFString(AcctPosOk, "True", "False");
                }
                // NR 事件状态：仅在当日存在收缩事件时附加
                String nrseg = "";
                If (NrHasEvent) {
                    nrseg = " NRH=" + Text(NrHighLv, 1) + " NRL=" + Text(NrLowLv, 1)
                        + " nrF=" + IIFString(NrFired, "True", "False");
                }
                WLog("DAILY " + symseg
                    + " N=" + Text(NVal, 1) + " EL=" + Text(EntryLongLv, 1) + " XL=" + Text(ExitLongLv, 1)
                    + " ES=" + Text(EntryShortLv, 1) + " XS=" + Text(ExitShortLv, 1)
                    + " units=" + Text(UnitCount) + " lots=" + Text(PLots) + " stop=" + Text(StopLine, 1)
                    + " equity=" + Text(CurEquity(), 0) + " win=" + Text(WinSize())
                    + " d1=" + d1s + d0seg + pendseg + acctseg + nrseg);
            }

            // ---- 5. 结束摘要：回测整段跑完的单行对账（只回测输出）----
            If (IsBacktestMode()) {
                If (BarStatus() == 2 And FinalDone == False) {
                    FinalDone = True;          // 幂等守卫：只打一次，OnInit 复位
                    // 与 DAILY 同守卫：无日线图层时不碰 Data1（访问非法）→ d1e 保持空、字段自然省略
                    String d1e = "";
                    If (DataCount() >= 2) {
                        d1e = " d1=" + Text(D1Close(1), 1);
                    }
                    WLog("=== 运行结束 BarCnt=" + Text(BarCnt) + " win=" + Text(WinSize()) + " DataCount=" + Text(DataCount())
                        + " N=" + Text(NVal, 1) + " pos=" + Text(MarketPosition()) + " contracts=" + Text(CurrentContracts())
                        + " lots=" + Text(PLots) + " equity=" + Text(CurEquity(), 0) + d1e + " ===");
                }
            }
        }
    }

    // data0 每根收盘驱动；诊断日志只在模拟/实盘实时输出
    OnBarClose(ArrayRef<Integer> indexs) {
        // [红线] 只放 Range[0:0]
        Range[0:0] {
            If (IsStrategyTradeMode()) {
                If (QuoteStatus() == Enum_QuoteStatus_RealTime) {
                    If (ArrayFind(indexs, 0)) {
                        LogBarHeartbeat();
                    }
                    LogData1Bar();
                }
            }
        }
    }

    // ==================== 回报事件 ====================
    // 回测不触发；OnOrder 维护挂单门禁，OnFill 仅日志并延后对账 2 根。

    OnFill(FillRef ordFill) {
        FillCnt = FillCnt + 1;
        // 成交后账户持仓可能滞后：延后 2 根新 bar 再对账
        SyncBlockBars = Max(SyncBlockBars, 2);
        WLogEvt("FILL #" + Text(FillCnt)
            + " id=" + Text(ordFill.orderId)
            + " fillId=" + ordFill.fillId
            + " side=" + IIFString(ordFill.side == Enum_Buy, "Buy", "Sell")
            + " off=" + IIFString(ordFill.combOffset == Enum_Entry, "Entry", "Exit")
            + " vol=" + Text(ordFill.volume)
            + " fillVol=" + Text(ordFill.fillVolume)
            + " price=" + Text(ordFill.price, 1)
            + " fillPrice=" + Text(ordFill.fillPrice, 1)
            + " lots=" + Text(PLots) + " units=" + Text(UnitCount)
            + " stop=" + Text(StopLine, 1));
    }

    OnOrder(OrderRef ord) {
        OrderCnt = OrderCnt + 1;
        // 挂单门禁维护：活跃状态入集合；其余（Filled/Canceled/Deleted/未知）视为终态移除
        Integer oid = ord.orderId;
        String ptxt = "";
        If (OrderIsPending(ord.status)) {
            If (PendingIndex(oid) < 0) {
                PendingAdd(oid);
                ptxt = " PEND_ADD";
            }
        }
        Else {
            If (PendingIndex(oid) >= 0) {
                PendingRemove(oid);
                ptxt = " PEND_DEL";
            }
        }
        WLogEvt("ORDER #" + Text(OrderCnt)
            + " id=" + Text(ord.orderId)
            + " status=" + Text(ord.status)
            + " side=" + IIFString(ord.side == Enum_Buy, "Buy", "Sell")
            + " off=" + IIFString(ord.combOffset == Enum_Entry, "Entry", "Exit")
            + " vol=" + Text(ord.volume)
            + " fvol=" + Text(ord.fillVolume)
            + " px=" + Text(ord.price, 1)
            + " lots=" + Text(PLots) + " units=" + Text(UnitCount)
            + " pending=" + Text(GetArraySize(PendingIds)) + ptxt);
    }

    // 停止策略单元时汇总事件到达情况（回测不输出，保持回归日志不变）
    OnExit() {
        If (StrategyMode() == Enum_Mode_Strategy_Trade()) {
            WLogEvt("EXIT fill=" + Text(FillCnt) + " order=" + Text(OrderCnt)
                + " pending=" + Text(GetArraySize(PendingIds)) + " ids=" + PendingStr());
        }
    }
